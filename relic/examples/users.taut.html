7// A user id is any segment that is not empty. A type is read from a path segment by its
8// `from_path`, written next to it as here; a type of another package is wrapped in a
9// newtype like this one to give it one.
10struct UserId { raw: Text }
11
12fn UserId::from_path(segment: Text): Option(UserId) {
13  if segment == "" { None } else { Some(UserId { raw: segment }) }
14}

16// Who may administer users. A handler that needs it performs `admin()`;
17// the `auth` middleware answers it, and only then reads `authorization`.
18struct Admin { token: Text }
19
20effect admin(): Admin
21
22fn bearer(v: Text): Option(Text) {
23  if Text::slice(
24    v,
25    0,
26    7
27  ) == Some("Bearer ") && Text::len(v) > 7 { Text::slice(v, 7, Text::len(v)) } else { None }
28}
29
30fn auth<!e>(k: () -> Response ! {admin} ∪ e): Response ! {get_headers} ∪ e {
31  handle { k() } with {
32    admin(), cont => match perform get_headers("authorization") {
33      None => text_response(Status::unauthorized(), "unauthorized")
34      Some(v) => match bearer(v) {
35        None => text_response(Status::forbidden(), "forbidden")
36        Some(t) => cont(Admin { token: t })
37      }
38    }
39  }
40}

66fn create(): Response ! {admin} {

71fn user(u: { user_id: UserId }): Response {
72  text_response(Status::ok(), u.user_id.raw)

75fn remove(u: { user_id: UserId }): Response ! {admin} {
76  let a := perform admin()
77  text_response(Status::ok(), concat(u.user_id.raw, concat(" removed by ", a.token)))
