6struct Todo { id: Nat, title: Text }
7
8fn length(xs: List(Todo)): Nat {
9  match xs {
10    Nil => 0
11    Cons(t, rest) => Succ(length(rest))
12  }
13}
14
15fn line(t: Todo): Text {
16  concat(Text::of_nat(t.id), concat(" ", t.title))
17}
18
19fn show(xs: List(Todo)): Text {
20  match xs {
21    Nil => ""
22    Cons(t, Nil) => line(t)
23    Cons(t, rest) => concat(line(t), concat("\n", show(rest)))
24  }
25}
26
27fn add(xs: List(Todo), title: Text): List(Todo) {
28  Cons(Todo { id: Succ(length(xs)), title: title }, xs)
29}
30
31fn remove(xs: List(Todo), id: Nat): List(Todo) {
32  match xs {
33    Nil => Nil
34    Cons(t, rest) => if t.id == id { remove(rest, id) } else { Cons(t, remove(rest, id)) }
35  }
36}
37
38fn sample(): List(Todo) {
39  Cons(Todo { id: 1, title: "buy milk" }, Cons(Todo { id: 2, title: "walk the dog" }, Nil))
40}
41
42fn list_todos(): Response {
43  text_response(Status::ok(), show(sample()))
44}
45
46// Without `?title` the item is "untitled".
47fn add_todo(added: { title: Option(Text) }): Response {
48  text_response(Status::created(), show(add(sample(), Option::unwrap_or(added.title, "untitled"))))
49}
50
51fn delete_todo(item: { id: Nat }): Response {
52  text_response(Status::ok(), show(remove(sample(), item.id)))
53}
54
55fn missing(): Response {
56  text_response(Status::not_found(), "not found")
57}
