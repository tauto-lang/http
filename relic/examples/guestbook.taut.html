6struct Entry { name: Text, message: Text }
7
8fn line(e: Entry): Text {
9  concat(e.name, concat(": ", e.message))
10}
11
12fn page(xs: List(Entry)): Text {
13  match xs {
14    Nil => ""
15    Cons(e, Nil) => line(e)
16    Cons(e, rest) => concat(line(e), concat("\n", page(rest)))
17  }
18}
19
20fn sign(xs: List(Entry), name: Text, message: Text): List(Entry) {
21  Cons(Entry { name: name, message: message }, xs)
22}
23
24fn book(): List(Entry) {
25  Cons(
26    Entry { name: "ada", message: "hello" },
27    Cons(Entry { name: "bea", message: "good morning" }, Nil)
28  )
29}
30
31fn read_book(): Response {
32  text_response(Status::ok(), page(book()))
33}
34
35fn sign_book(note: { message: Text, name: Text }): Response {
36  text_response(Status::created(), page(sign(book(), note.name, note.message)))
37}
38
39fn missing(): Response {
40  text_response(Status::not_found(), "not found")
41}
