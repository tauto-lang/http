4// What a handler asks of the request. `handler` from `router!` answers all
5// three with `answer_request`, around the middleware and the handler.
6// `request_body` gives the body as `Bytes`, made from the request's body text
7// each time it is asked.
8// `get_headers` reads one request header (the first, when the request has several of that
9// name). `set_header` adds a response header; a `content-length`, a `transfer-encoding`
10// or (in `finish_response`) a `connection` set this way is dropped when the response is
11// written (`response.taut`): relic decides how the body is framed and whether the
12// connection stays open.
13pub effect get_headers(name: Text): Option(Text) resume 1
14pub effect set_header(name: Text, value: Text): Unit resume 1
15pub effect request_body(): Bytes resume 1
16
17// `k` answers `request`; the headers it sets go before the ones its response has.
18pub fn answer_request<!e>(
19  request: HttpRequest,
20  k: () -> Response ! {get_headers, set_header, request_body} ∪ e
21): Response ! e {
22  var added := Nil
23  let out := handle { k() } with {
24    get_headers(name), cont => cont(HttpRequest::header(request, name))
25    set_header(name, value), cont => {
26      added := Cons(#(@name name, @value value), added)
27      cont(#())
28    }
29    request_body(), cont => cont(Bytes::of_text(HttpRequest::body(request)))
30  }
31  prepend_headers(out, added)
32}
