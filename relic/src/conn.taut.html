54// HTTP/1.1 over TCP (`--allow-net`). The bytes of a connection go through the
55// readers of `frame.taut` into a `HttpRequest`, a function answers it, and the
56// response goes back on the same connection, which stays open for the next request
57// until the client asks to close, goes quiet, or has had its fill (`requests_per_connection`).
58// Reading a request is pure (`take_all`), so it runs without a network.
59
60// The most bytes a request's head may take.
61fn head_limit(): Nat {
62  16384
63}
64
65// The most bytes a request's body may take; past it the answer is 413.
66fn body_limit(): Nat {
67  1048576
68}
69
70// The most bytes a chunk's size line may take, with its extensions, and the
71// trailers together.
72fn size_line_limit(): Nat {
73  1024
74}
75
76// Where a request stands as its bytes arrive.
77inductive Reading: Type {
78  ReadingHead(h: RequestHead): Reading
79  // The request without its body, how many body bytes are still to come, and
80  // the pieces of the body so far, the last first.
81  ReadingSized(r: HttpRequest, left: Nat, parts: List(List(Nat))): Reading
82  // The request without its body, the chunked reader, the pieces of the body
83  // so far, the last first, and how many bytes they hold.
84  ReadingChunks(r: HttpRequest, c: Chunks, parts: List(List(Nat)), size: Nat): Reading
85}

87pub(open) inductive Took: Type {
88  // More bytes are needed.
89  Wants(r: Reading): Took
90  // The whole request, and the bytes after it.
91  Whole(r: HttpRequest, rest: List(Nat)): Took
92  // The request cannot be read; the answer has this status.
93  RefusedWith(status: Status): Took
94}
95
96pub fn reading_start(): Reading {
97  ReadingHead(Head::start(head_limit()))
98}

100fn with_text(r: HttpRequest, o: Option(Text), rest: List(Nat)): Took {
101  match o {
102    None => RefusedWith(Status::bad_request())
103    Some(t) => ⊢ Took from match r {
104      Mk(m, p, q, hs, b) => Whole(HttpRequest::Mk(m, p, q, hs, t), rest)
105    }
106  }
107}

109// The whole request with its body; a body that is not UTF-8 is refused.
110fn with_body(r: HttpRequest, body: List(List(Nat)), rest: List(Nat)): Took {
111  with_text(r, Text::of_utf8(List::concat(List::reverse(body))), rest)
112}
113
114fn take_sized(r: HttpRequest, left: Nat, parts: List(List(Nat)), bs: List(Nat)): Took {
115  match feed(length_step, left, bs) {
116    Fed::Mk(n, out, rest) => ⊢ Took from match n {
117      Zero => with_body(r, Cons(out, parts), rest)
118      Succ(k) => Wants(ReadingSized(r, n, Cons(out, parts)))
119    }
120  }
121}

123pub module Chunks {
124  // `Some(true)` once the body ended, `Some(false)` when it is malformed.
125  fn over(c: Chunks): Option(Bool) {
126    match c {
127      ChunkSize(n, any, room) => None
128      ChunkExt(n, room) => None
129      ChunkSizeLf(n) => None
130      ChunkData(left) => None
131      ChunkDataLf => None
132      TrailerStart(room) => None
133      TrailerLine(room) => None
134      TrailerLf(room) => None
135      ChunksEndLf => None
136      ChunksDone => Some(true)
137      ChunksBad => Some(false)
138    }
139  }
140
141  pub fn took(r: HttpRequest, c: Chunks, parts: List(List(Nat)), size: Nat, rest: List(Nat)): Took {
142    match Nat::le(size, body_limit()) {
143      false => RefusedWith(Status::content_too_large())
144      true => ⊢ Took from match Chunks::over(c) {
145        None => Wants(ReadingChunks(r, c, parts, size))
146        Some(y) => ⊢ Took from match y {
147          true => with_body(r, parts, rest)
148          false => RefusedWith(Status::bad_request())
149        }
150      }
151    }
152  }
153}
154
155fn take_chunks(r: HttpRequest, c: Chunks, parts: List(List(Nat)), size: Nat, bs: List(Nat)): Took {
156  match Chunks::feed(size_line_limit(), c, bs) {
157    Fed::Mk(c2, out, rest) => Chunks::took(
158      r,
159      c2,
160      Cons(out, parts),
161      Nat::add(size, List::length(out)),
162      rest
163    )
164  }
165}

167// After the head: how the body is framed decides how to read it. A request
168// never runs to the end of the connection.
169fn take_body(r: HttpRequest, rest: List(Nat)): Took {
170  match HttpRequest::framing(body_limit(), HttpRequest::headers(r)) {
171    NoBody => Whole(r, rest)
172    Sized(n) => take_sized(r, n, Nil, rest)
173    Chunked => take_chunks(r, Chunks::start(size_line_limit()), Nil, Zero, rest)
174    ToClose => RefusedWith(Status::bad_request())
175    Refused(status) => RefusedWith(status)
176  }
177}

179pub module Head {
180  pub fn parsed(o: Option(HttpRequest), rest: List(Nat)): Took {
181    match o {
182      None => RefusedWith(Status::bad_request())
183      Some(r) => take_body(r, rest)
184    }
185  }

188fn take_head(h: RequestHead, bs: List(Nat)): Took {
189  took_head(Head::feed_fast(h, bs))
190}

192fn head_rest(f: Fed(RequestHead)): List(Nat) {
193  match f {
194    Fed::Mk(s, out, rest) => rest
195  }
196}

198// A new request's head read from the received bytes where they are: its lines
199// as positions (`Head::read_at_eq`), and the request read from them
200// (`request_at`), as `took_head(Head::feed_at(h, raw, 0, n))` reads it.
201fn took_head_at(h: RequestHead, raw: Bytes, n: Nat): Took {
202  match h {
203    HeadReading(line, done, limit) => ⊢ Took from match line {
204      Nil => ⊢ Took from match done {
205        Nil => ⊢ Took from match Head::read_at(raw, 0, 0, true, false, Nil, limit, n) {
206          AtRead(spans, next, k) => Head::parsed(
207            lines_at(raw, List::reverse(spans)),
208            head_rest(Head::feed_at(HeadRead(Nil), raw, next, k))
209          )
210          AtReading(s, i, empty, blank, spans, left) => took_head(
211            Head::at_fed(raw, AtReading(s, i, empty, blank, spans, left))
212          )
213          AtTooLong(next, k) => took_head(Head::at_fed(raw, AtTooLong(next, k)))
214        }
215        Cons(d, ds) => took_head(Head::feed_at(h, raw, 0, n))
216      }
217      Cons(b, bs) => took_head(Head::feed_at(h, raw, 0, n))
218    }
219    HeadRead(ls) => took_head(Head::feed_at(h, raw, 0, n))
220    HeadTooLong => took_head(Head::feed_at(h, raw, 0, n))
221  }
222}
223
224// What the head reader's result means for the request.
225fn took_head(f: Fed(RequestHead)): Took {
226  match f {
227    Fed::Mk(s, out, rest) => ⊢ Took from match s {
228      HeadReading(line, done, limit) => Wants(ReadingHead(s))
229      HeadRead(ls) => Head::parsed(parse_lines(ls), rest)
230      HeadTooLong => RefusedWith(Status::request_header_fields_too_large())
231    }
232  }
233}

235// The bytes that arrived next, wherever the request stands.
236fn take(rd: Reading, bs: List(Nat)): Took {
237  match rd {
238    ReadingHead(h) => take_head(h, bs)
239    ReadingSized(r, left, parts) => take_sized(r, left, parts, bs)
240    ReadingChunks(r, c, parts, size) => take_chunks(r, c, parts, size, bs)
241  }
242}

244// The bytes sent, as they arrived, from the start of a connection.
245pub fn take_all(rd: Reading, pieces: List(List(Nat))): Took {
246  match pieces {
247    Nil => Wants(rd)
248    Cons(bs, more) => ⊢ Took from match take(rd, bs) {
249      Wants(rd2) => take_all(rd2, more)
250      Whole(r, rest) => Whole(r, rest)
251      RefusedWith(status) => RefusedWith(status)
252    }
253  }
254}

256// The bytes on the wire to what was read, if anything was. The answer to
257// `HEAD` has no body, and every answer closes the connection.
258pub fn answer_took<!e>(f: (HttpRequest) -> Response ! e, t: Took): Option(Bytes) ! e {
259  match t {
260    Wants(rd) => None
261    Whole(r, rest) => Some(finish_response(is_head(r), true, f(r)))
262    RefusedWith(status) => Some(refusal(status))
263  }
264}

266fn refusal(status: Status): Bytes {
267  finish_response(false, true, text_response(status, reason(status)))
268}

277// stream and gives it back.

280// A connection given back with whether the write reached it.
281struct(affine) Wrote { conn: TcpStream, ok: Bool }

283// Reads from `c` until the request is whole or cannot be read, starting from
284// `rd` (`reading_start()` for a new connection). A read that times out gives
285// `RefusedWith(408)`; `Wants` once the client stops sending or the read fails
286// for another reason.
287pub fn read_request(c: TcpStream, rd: Reading, deadline: i64): Got ! {tcp_read, div} {
288  match perform tcp_read(c, 16384, Some(deadline)) {
289    TcpRead { stream, got } => match got {
290      Err(e) => Got {
291        conn: stream,
292        took: match e {
293          TimedOut => RefusedWith(Status::request_timeout())
294          Denied(m) => Wants(rd)
295          Failed(m) => Wants(rd)
296        }
297      }
298      Ok(raw) => match rd {
299        // The head is read from the received bytes where they are
300        // (`Head::feed_at_eq`: the same as `take_head` on their list).
301        ReadingHead(h) => match Bytes::len(raw) == 0 {
302          true => Got { conn: stream, took: Wants(rd) }
303          false => match took_head_at(h, raw, Nat::of_u64(Bytes::len(raw))) {
304            Wants(rd2) => read_request(stream, rd2, deadline)
305            Whole(r, rest) => Got { conn: stream, took: Whole(r, rest) }
306            RefusedWith(status) => Got { conn: stream, took: RefusedWith(status) }
307          }
308        }
309        ReadingSized(r, left, parts) => read_rest(stream, rd, deadline, raw)
310        ReadingChunks(r, c, parts, size) => read_rest(stream, rd, deadline, raw)
311      }
312    }
313  }
314}
315
316fn read_rest(c: TcpStream, rd: Reading, deadline: i64, raw: Bytes): Got ! {tcp_read, div} {
317  match Bytes::to_list(raw) {
318    Nil => Got { conn: c, took: Wants(rd) }
319    Cons(b, more) => match take(rd, nats(Cons(b, more))) {
320      Wants(rd2) => read_request(c, rd2, deadline)
321      Whole(r, rest) => Got { conn: c, took: Whole(r, rest) }
322      RefusedWith(status) => Got { conn: c, took: RefusedWith(status) }
323    }
324  }
325}

352// Whether a `connection` value lists the `close` option (RFC 9110 7.6.1).
353fn lists_close(parts: List(List(Nat))): Bool {
354  match parts {
355    Nil => false
356    Cons(p, more) => bytes_eq_ci(trim_spaces(p), Text::bytes("close")) || lists_close(more)
357  }
358}

360fn drop_spaces(l: List(Nat)): List(Nat) {
361  match l {
362    Nil => Nil
363    Cons(b, more) => match b == 32 || b == 9 {
364      true => drop_spaces(more)
365      false => l
366    }
367  }
368}
369
370fn trim_spaces(l: List(Nat)): List(Nat) {
371  List::reverse(drop_spaces(List::reverse(drop_spaces(l))))
372}
373
374fn split_parts(o: List(List(Nat)) × List(Nat)): List(List(Nat)) {
375  match o {
376    #(parts, last) => Cons(last, parts)
377  }
378}

380// Whether any of the `connection` values lists `close`.
381fn any_lists_close(vs: List(Text)): Bool {
382  match vs {
383    Nil => false
384    Cons(v, more) => lists_close(
385      split_parts(Bytes::split(Text::bytes(v), 44))
386    ) || any_lists_close(more)
387  }
388}

390// Whether the client asked to close the connection after this request: any of its
391// `connection` headers, not only the first, lists `close`.
392pub fn asks_close(r: HttpRequest): Bool {
393  any_lists_close(HttpRequest::header_all(r, "connection"))
394}

396// When a read or a write that starts now gives up. A clock too near the end
397// of `i64` to add the wait to gives up at once.
398fn deadline(): i64 ! {monotonic_clock} {
399  let now := perform monotonic_clock()
400  handle { now + wait_ns() } with { overflow() => now }
401}

403fn finish(c: TcpStream): Unit ! {tcp_shutdown, tcp_close} {
404  match perform tcp_shutdown(TcpStream::stopper(&c), ShutdownWrite) {
405    Ok(v) => #()
406    Err(e) => #()
407  }
408  perform tcp_close(c)
409}

411// The next request on a connection: what is left of the bytes already
412// received, then what arrives.
413fn next_request(c: TcpStream, rest: List(Nat)): Got ! {tcp_read, monotonic_clock, div} {
414  let deadline := deadline()
415  match rest {
416    Nil => read_request(c, reading_start(), deadline)
417    Cons(b, more) => match take(reading_start(), rest) {
418      Wants(rd) => read_request(c, rd, deadline)
419      Whole(r, rest2) => Got { conn: c, took: Whole(r, rest2) }
420      RefusedWith(status) => Got { conn: c, took: RefusedWith(status) }
421    }
422  }
423}
424
425fn write_bytes(c: TcpStream, b: Bytes): Wrote ! {tcp_write, monotonic_clock, stderr} {
426  match perform tcp_write(c, b, Some(deadline())) {
427    TcpWritten { stream, result } => match result {
428      Ok(u) => Wrote { conn: stream, ok: true }
429      Err(e) => {
430        perform stderr(io_message(e))
431        Wrote { conn: stream, ok: false }
432      }
433    }
434  }
435}

469// One connection, answered by `f`: its requests one after another, up to `left`.
470fn answer_conn<!h: Host>(
471  c: TcpStream,
472  f: (HttpRequest) -> Response ! h,
473  rest: List(Nat),
474  left: u64
475): Unit ! {ConnIo} ∪ h {
476  match next_request(c, rest) {
477    Got { conn, took } => match took {
478      Wants(rd) => finish(conn)
479      RefusedWith(status) => match write_bytes(conn, refusal(status)) {
480        Wrote { conn, ok } => finish(conn)
481      }
482      Whole(r, rest2) => {
483        let last := asks_close(r) || left <= 1
484        // The connection waits in a mutex while `f` runs: `f` may resume more than
485        // once, and a connection may not be held across that.
486        let parked := perform mutex_new(conn)
487        let bytes := finish_response(is_head(r), last, f(r))
488        match write_bytes(Mutex::take(parked), bytes) {
489          Wrote { conn, ok } => match ok && not(last) {
490            true => answer_conn(conn, f, rest2, handle { left - 1 } with { overflow() => 0 })
491            false => finish(conn)
492          }
493        }
494      }
495    }
496  }
497}

499fn answer_loop<!h: Host>(
500  l: TcpListener,
501  f: (HttpRequest) -> Response ! h,
502  slots: Semaphore
503): Unit ! {ServeIo, ∀ T. spawn(() -> T ! {TaskIo} ∪ h)} ∪ h {
504  Semaphore::acquire(slots)
505  match perform tcp_accept(l, None) {
506    TcpAccepted { listener, got } => match got {
507      Err(Denied(m2)) => perform stderr(concat(m2, "\n"))
508      Err(e) => {
509        Semaphore::release(slots)
510        perform stderr(concat(io_message(e), "\n"))
511        perform sleep(accept_retry_ms())
512        answer_loop(listener, f, slots)
513      }
514      Ok(c) => {
515        Task::detach(perform spawn(() => {
516          answer_conn(c, f, Nil, requests_per_connection())
517          Semaphore::release(slots)
518        }))
519        answer_loop(listener, f, slots)
520      }
521    }
522  }
523}

525// Listen on `address` and answer each request with `f`. Connections are
526// served at once, each in a task, at most `connection_limit()` of them; the
527// next is accepted once one closes. They stay open for further requests
528// (HTTP/1.1 persistent connections) until the client closes or asks to
529// close, or is idle past `wait_ns`. The handler's state is an effect it
530// performs, not a value this loop holds. Stops when listening fails or
531// accepting is not permitted.
532pub fn serve_response<!h: Host>(
533  address: Text,
534  f: (HttpRequest) -> Response ! h
535): Unit ! {ServeIo, ∀ T. spawn(() -> T ! {TaskIo} ∪ h)} ∪ h {
536  match perform tcp_listen(address) {
537    Err(e) => perform stderr(io_message(e))
538    Ok(l) => {
539      perform stderr(concat("listening on ", TcpListener::address(&l)))
540      answer_loop(l, f, Semaphore::new(connection_limit()))
541    }
542  }
