26// ---- `tauto fmt` ------------------------------------------------------------------------------
27// One entry a line, indented two spaces for each `{` of a route's children, a blank line kept
28// where the table has one. A text with a comment is left as written.
29
30pub fn fmt_pad(n: u64): Text ! {div, overflow} {
31  if n == 0 { "" } else { concat("  ", fmt_pad(n - 1)) }
32}

34pub fn fmt_kids(st: List(Text)): u64 ! {div, overflow} {
35  match st {
36    Nil => 0
37    Cons(k, rest) => if k == "kids" { 1 + fmt_kids(rest) } else { fmt_kids(rest) }
38  }
39}

41pub fn fmt_plain(ts: List(SynTok)): Bool ! {div, overflow} {
42  match ts {
43    Nil => true
44    Cons(t, rest) => if t.skipped && t.kind != "Newline" && t.kind != "Whitespace" {
45      false
46    } else {
47      fmt_plain(rest)
48    }
49  }
50}

52// `p` is the token before (`/` for a path; `}cap`, `}row` or `}kids` for a `}`), `tight` that no
53// space goes before
54// the next token, `nl` the line breaks since it and `st` the braces open: `cap` for a capture,
55// `row` for the row and `kids` for the children of a route.
56pub fn fmt_go(
57  ts: List(SynTok),
58  p: Text,
59  tight: Bool,
60  nl: u64,
61  st: List(Text)
62): Text ! {div, overflow} {
63  match ts {
64    Nil => ""
65    Cons(t, rest) => if t.skipped {
66      fmt_go(rest, p, tight, if t.kind == "Newline" { nl + 1 } else { nl }, st)
67    } else {
68      let x := t.text
69      let path := t.kind == "Literal" || t.kind == "CatchAll"
70      let top := match st {
71        Nil => ""
72        Cons(k, _) => k
73      }
74      if x == "{" {
75        let kind := if p == "/" { "cap" } else if p == "!" { "row" } else { "kids" }
76        concat(
77          if kind == "cap" { "" } else { " " },
78          concat("{", fmt_go(rest, "{", kind == "cap", 0, Cons(kind, st)))
79        )
80      } else if x == "}" {
81        let below := match st {
82          Nil => Nil
83          Cons(_, rest_st) => rest_st
84        }
85        let prefix := if top == "cap" {
86          ""
87        } else if top == "row" {
88          if p == "{" { "" } else { " " }
89        } else if p == "{" {
90          " "
91        } else {
92          concat("\n", fmt_pad(fmt_kids(below)))
93        }
94        concat(prefix, concat("}", fmt_go(rest, concat("}", top), false, 0, below)))
95      } else {
96        let start := (x == "middleware" && p != ":" && p != ",") ||
97        (path && p != "}cap")
98        let prefix := if p == "" {
99          ""
100        } else if start {
101          concat(if nl > 1 && p != "{" { "\n\n" } else { "\n" }, fmt_pad(fmt_kids(st)))
102        } else if x == "," || x == ":" || x == "]" || x == "&" || tight || (path && p == "}cap") {
103          ""
104        } else {
105          " "
106        }
107        concat(
108          prefix,
109          concat(
110            x,
111            fmt_go(rest, if path { "/" } else { x }, x == "?" || x == "[" || x == "&", 0, st)
112          )
113        )
114      }
115    }
116  }
117}

119pub fn router_fmt(ts: List(SynTok), width: u64): Option(Text) ! {div, overflow} {
120  if fmt_plain(ts) { Some(concat(fmt_go(ts, "", false, 0, Nil), "\n")) } else { None }
121}

123// ---- the colours ------------------------------------------------------------------------------
124// A path is a string, `middleware` and a method are keywords, the name before a `:` of a capture or
125// a query is a property, what a method or `middleware` names is a function, a type after a `:` is
126// a type, and the route's own name is an enum member.
127
128pub fn tokens_colon(ts: List(SynTok)): Bool {
129  match ts {
130    Nil => false
131    Cons(t, rest) => if t.skipped { tokens_colon(rest) } else { t.text == ":" }
132  }
133}

135pub fn tokens_method(x: Text): Bool {
136  x == "get" || x == "post" || x == "put" || x == "patch" || x == "delete" || x == "head" ||
137  x == "options" || x == "any"
138}

140// `after` is what the `:` before this token follows: `kw` for `middleware` or a method, else `name`.
141pub fn tokens_go(ts: List(SynTok), prev: Text, after: Text): List(SynHighlight) ! {div} {
142  match ts {
143    Nil => Nil
144    Cons(t, rest) => if t.skipped {
145      tokens_go(rest, prev, after)
146    } else {
147      let x := t.text
148      let colon := tokens_colon(rest)
149      let kind := if t.kind == "Literal" || t.kind == "CatchAll" {
150        "string"
151      } else if colon && (x == "middleware" || tokens_method(x)) {
152        "keyword"
153      } else if x == "!" || x == "?" || x == "&" || x == "{" || x == "}" || x == "[" || x == "]" ||
154      x == "," || x == ":" {
155        "operator"
156      } else if colon {
157        "property"
158      } else if (prev == ":" || prev == ",") && after == "kw" {
159        "function"
160      } else if prev == ":" {
161        "type"
162      } else {
163        "enumMember"
164      }
165      let next_after := if kind == "keyword" {
166        "kw"
167      } else if kind == "operator" || kind == "function" {
168        after
169      } else {
170        "name"
171      }
172      Cons(SynHighlight { start: t.start, end: t.end, kind: kind }, tokens_go(rest, x, next_after))
173    }
174  }
175}

177pub fn router_tokens(ts: List(SynTok)): List(SynHighlight) ! {div} {
178  tokens_go(ts, "", "name")
179}

181// `router!` reads a route table and writes `enum Route`, `fn route` and
182// `fn handler`. `route` looks at the path and the query only, and performs
183// nothing. `handler` answers `get_headers`, `set_header` and
184// `request_body`, runs the middleware from the outside in, then the method.
185//
186// The first line names the enum and the row `handler` is left with: the
187// effects the middleware and the handlers still perform once the middleware
188// has handled what it handles. `! {}` is the empty row.
189//
190// A route is a path, an optional query, the name of its constructor, its
191// methods and, in braces, its children and their middleware. A capture
192// `{user_id: UserId}` is one whole segment; `?page: Nat` must be present and
193// `?[title: Text]` is an `Option(Text)`. `Text`, `Nat`, `u64` and `i64` are
194// read by `path.taut`; another type `T` by `T::from_path`. `/*` is the last
195// route at the top, and its methods cover every method.
196syntax!(router_syntax, ```par
197%start Router
198%title "router"
199%%
200Router: Name Row { Entry };
201Row: Bang LBrace [ Names ] RBrace;
202Names: Name { CommaName };
203CommaName: Comma Name;
204Entry: Middleware | Route;
205Middleware: MiddlewareKw Colon Names;
206Route: Path [ Query ] Name { Verb } [ Children ];
207Path: CatchAll | Pattern;
208Pattern: Literal { PathPart };
209PathPart: Capture | Literal;
210Capture: LBrace Name Colon Name RBrace;
211Query: Question QueryItem { AmpItem };
212AmpItem: Amp QueryItem;
213QueryItem: Field | OptionalField;
214OptionalField: LBracket Field RBracket;
215Field: Name Colon Name;
216Verb: Method Colon Name;
217Children: LBrace { Entry } RBrace;
218MiddlewareKw: 'middleware';
219Method: 'get' | 'post' | 'put' | 'patch' | 'delete' | 'head' | 'options' | 'any';
220CatchAll: '/*';
221Question: '?';
222Bang: '!';
223Amp: '&';
224Colon: ':';
225Comma: ',';
226LBrace: '{';
227RBrace: '}';
228LBracket: '[';
229RBracket: ']';
230Literal: /\/[A-Za-z0-9_.~\/-]*/;
231Name: /[A-Za-z_][A-Za-z0-9_]*/;
232```, with(fmt: router_fmt, tokens: router_tokens)) =>
233  struct AmpItem, struct Capture, struct Children, struct CommaName, enum Entry, struct Field,
234  enum Method, struct Middleware, struct Names, struct OptionalField, enum Path, enum PathPart,
235  struct Pattern, struct Query, enum QueryItem, struct Route, struct Router, struct Row,
236  struct Verb, fn parse

253fn last_byte_is(t: Text, b: u64): Bool {
254  let n := Text::len(t)
255  if n == 0 {
256    false
257  } else {
258    match Text::byte(t, handle { n - 1 } with { overflow() => 0 }) {
259      Some(c) => U64::of_u8(c) == b
260      None => false
261    }
262  }
263}

265fn first_byte_is(t: Text, b: u64): Bool {
266  match Text::byte(t, 0) {
267    Some(c) => U64::of_u8(c) == b
268    None => false
269  }
270}

322// The segments of a path. A capture is one whole segment: the text before
323// it ends with `/`, and the text after it starts with `/`.
324fn parts_segs(
325  ps: List(router_syntax::PathPart),
326  after_slash: Bool
327): List(Seg) ! {macro_error, div} {
328  match ps {
329    Nil => Nil
330    Cons(p, rest) => match p {
331      router_syntax::PathPart::Literal(tok) => if after_slash || first_byte_is(tok.text, 47) {
332        Syn::append(lit_segs(tok.text), parts_segs(rest, last_byte_is(tok.text, 47)))
333      } else {
334        fail(tok.span, "router: a capture is a whole segment; write `/` after it")
335      }
336      router_syntax::PathPart::Capture(c) => if after_slash {
337        Cons(Param(c.name, c.name_2), parts_segs(rest, false))
338      } else {
339        fail(c.span, "router: a capture is a whole segment; write `/` before it")
340      }
341    }
342  }
343}

345fn pattern_segs(p: router_syntax::Pattern): List(Seg) ! {macro_error, div} {
346  Syn::append(lit_segs(p.literal.text), parts_segs(p.path_part, last_byte_is(p.literal.text, 47)))
347}

436fn insert_slot(s: Slot, ss: List(Slot), at: SynSpan): List(Slot) ! {macro_error, div} {
437  match ss {
438    Nil => Cons(s, Nil)
439    Cons(t, rest) => if s.name == t.name {
440      fail(at, concat("router: a field twice: ", s.name))
441    } else if Text::compare(s.name, t.name) < 0 {
442      Cons(s, ss)
443    } else {
444      Cons(t, insert_slot(s, rest, at))
445    }
446  }
447}

471fn lower(
472  r: router_syntax::Route,
473  prefix: List(Seg),
474  mw: List(SynToken),
475  model: SynIdent,
476  nested: Bool
477): List(Flat) ! {macro_error, div} {
478  let at := r.span
479  let catch_all := match r.path {
480    router_syntax::Path::CatchAll(t) => true
481    router_syntax::Path::Pattern(p) => false
482  }
483  if catch_all && nested { fail(at, "router: `/*` is only at the top") } else { #() }
484  let segs := match r.path {
485    router_syntax::Path::CatchAll(t) => prefix
486    router_syntax::Path::Pattern(p) => Syn::append(prefix, pattern_segs(p))
487  }
488  let kids := children_of(r.children)
489  let chain := Syn::append(mw, middleware_of(kids))
490  let vs := verbs_of(r.verb, Nil)
491  if has_any(vs) && List::length(vs) != 1 {
492    fail(at, "router: `any` is the only method")
493  } else { #() }
494  let qs := queries_of(r.query)
495  if catch_all {
496    match qs {
497      Nil => #()
498      Cons(q, rest) => fail(at, "router: `/*` has no query")
499    }
500    match kids {
501      Nil => #()
502      Cons(c, rest) => fail(at, "router: `/*` has no children")
503    }
504  } else { #() }
505  let below := lower_all(kids, segs, chain, model, true)
506  match vs {
507    Nil => match below {
508      Nil => fail(at, concat("router: write a method for ", r.name.text))
509      Cons(b, rest) => below
510    }
511    Cons(v, rest) => {
512      let slots := sort_slots(Syn::append(seg_slots(segs), query_slots(qs)), at)
513      Cons(
514        Flat {
515          ctor: Syn::decl_name(model, r.name.text),
516          slots: slots,
517          record: Syn::decl_name(model, concat("@record.", dotted(slots))),
518          segs: segs,
519          queries: qs,
520          middleware: chain,
521          verbs: vs,
522          catch_all: catch_all,
523          span: at
524        },
525        below
526      )
527    }
528  }
529}

531fn lower_all(
532  es: List(router_syntax::Entry),
533  prefix: List(Seg),
534  mw: List(SynToken),
535  model: SynIdent,
536  nested: Bool
537): List(Flat) ! {macro_error, div} {
538  match es {
539    Nil => Nil
540    Cons(e, rest) => match e {
541      router_syntax::Entry::Middleware(m) => lower_all(rest, prefix, mw, model, nested)
542      router_syntax::Entry::Route(r) => Syn::append(
543        lower(r, prefix, mw, model, nested),
544        lower_all(rest, prefix, mw, model, nested)
545      )
546    }
547  }
548}

550fn check_names(fs: List(Flat), seen: List(Text)): Unit ! {macro_error, div} {
551  match fs {
552    Nil => #()
553    Cons(f, rest) => if has_text(seen, f.ctor.name) {
554      fail(f.span, concat("router: route name twice: ", f.ctor.name))
555    } else {
556      check_names(rest, Cons(f.ctor.name, seen))

561fn check_tail(fs: List(Flat)): Unit ! {macro_error, div} {
562  match fs {
563    Nil => fail(Syn::call_span(), "router: write `/* Name any: handler` last")
564    Cons(f, Nil) => if f.catch_all { check_any(f) } else {
565      fail(f.span, "router: the last route is `/*`")
566    }
567    Cons(f, rest) => if f.catch_all {
568      fail(f.span, "router: `/*` is the last route")
569    } else {
570      check_tail(rest)
571    }
572  }
573}

660    Cons(s, rest) => Cons(ExRecord(f.record, record_fields(f.slots), at), Nil)
661  }
662}

664fn ctor_value(f: Flat, at: SynSpan): SynExpr {
665  match f.slots {
666    Nil => v(f.ctor)
667    Cons(s, rest) => app(f.ctor, record_args(f, at), at)
668  }
669}

671fn ctor_pat(f: Flat): SynPattern {
672  match f.slots {
673    Nil => PtCtor(f.ctor, Nil)
674    Cons(s, rest) => PtCtor(f.ctor, Cons(PtRecord(f.record, record_pats(f.slots)), Nil))
675  }
676}

678fn ctor_of(f: Flat, at: SynSpan): SynCtor {
679  match f.slots {
680    Nil => SynCtor { name: f.ctor, fields: Nil }
681    Cons(s, rest) => SynCtor {
682      name: f.ctor,
683      fields: Cons(#(None, ExApp(v(f.record), List::map((s: Slot) => s.ty, f.slots), at)), Nil)
684    }
685  }

689fn wrap(names: List(SynToken), inner: SynExpr, at: SynSpan): SynExpr {
690  match names {
691    Nil => inner
692    Cons(n, rest) => app(Syn::ident(n), Cons(lam(wrap(rest, inner, at), at), Nil), at)
693  }

697  app(verb.handler, record_args(f, at), at)

700fn method_ctor(m: Text): Text {
701  if m == "get" { "Get" } else if m == "post" { "Post" } else if m == "put" { "Put" } else if m == "delete" { "Delete" } else if m == "patch" { "Patch" } else if m == "head" { "Head" } else { "Options" }
702}
703
704fn allow_piece(m: Text): Text {
705  if m == "get" { "GET" } else if m == "post" { "POST" } else if m == "put" { "PUT" } else if m == "delete" { "DELETE" } else if m == "patch" { "PATCH" } else if m == "head" { "HEAD" } else { "OPTIONS" }
706}
707
708fn allow_one(m: Text, on: Bool): Text {
709  if on { allow_piece(m) } else { "" }
710}
711
712fn join_allow(a: Text, b: Text): Text {
713  if a == "" { b } else if b == "" { a } else { concat(a, concat(", ", b)) }
714}
715
716fn allow_text(vs: List(Verb)): Text {
717  let head := has_method(vs, "head") || has_method(vs, "get")
718  join_allow(
719    allow_one("get", has_method(vs, "get")),
720    join_allow(
721      allow_one("post", has_method(vs, "post")),
722      join_allow(
723        allow_one("put", has_method(vs, "put")),
724        join_allow(
725          allow_one("delete", has_method(vs, "delete")),
726          join_allow(
727            allow_one("patch", has_method(vs, "patch")),
728            join_allow(allow_one("head", head), allow_one("options", has_method(vs, "options")))
729          )
730        )
731      )
732    )
733  )
734}
735
736fn find_verb(vs: List(Verb), m: Text): Option(Verb) {
737  match vs {
738    Nil => None
739    Cons(v, rest) => if v.method == m { Some(v) } else { find_verb(rest, m) }
740  }
741}
742
743fn has_method(vs: List(Verb), m: Text): Bool {
744  match find_verb(vs, m) {
745    Some(v) => true
746    None => false
747  }
748}
749
750fn covered(vs: List(Verb)): Bool {
751  has_method(
752    vs,
753    "get"
754  ) && has_method(
755    vs,
756    "post"
757  ) && has_method(
758    vs,
759    "put"
760  ) && has_method(vs, "delete") && has_method(vs, "patch") && has_method(vs, "options")
761}
762
763fn method_arms(f: Flat, at: SynSpan): List(SynArm) {
764  let base := List::map(
765    (verb: Verb) => arm(
766      pctor(concat("Method::", method_ctor(verb.method)), Nil),
767      handler_call(verb, f, at)
768    ),
769    f.verbs
770  )
771  // Without `head`, `GET` answers `HEAD`; the body is not sent.
772  let with_head := match find_verb(f.verbs, "head") {
773    Some(h) => base
774    None => match find_verb(f.verbs, "get") {
775      None => base
776      Some(get) => Cons(arm(pctor("Method::Head", Nil), handler_call(get, f, at)), base)
777    }
778  }
779  if covered(f.verbs) {
780    with_head
781  } else {
782    Syn::append(
783      with_head,
784      Cons(
785        arm(PtWild, call("method_not_allowed", Cons(ExText(allow_text(f.verbs), at), Nil), at)),
786        Nil
787      )
788    )
789  }
790}
791
792fn dispatch(f: Flat, req: SynIdent, at: SynSpan): SynExpr {
793  match find_verb(f.verbs, "any") {
794    Some(verb) => handler_call(verb, f, at)
795    None => mat(call("HttpRequest::method", Cons(v(req), Nil), at), method_arms(f, at), at)
796  }
797}

801// Reading a captured text as the type `ty`.
802fn read_as(ty: SynToken, raw: SynExpr, at: SynSpan): SynExpr {
803  if ty.text == "Text" {
804    call("path_text", Cons(raw, Nil), at)
805  } else if ty.text == "Nat" {
806    call("path_nat", Cons(raw, Nil), at)
807  } else if ty.text == "u64" {
808    call("path_u64", Cons(raw, Nil), at)
809  } else if ty.text == "i64" {
810    call("path_i64", Cons(raw, Nil), at)
811  } else {
812    app(Syn::decl_name(Syn::ident(ty), concat(ty.text, "::from_path")), Cons(raw, Nil), at)
813  }
814}
815
816// `Some(x) => then`, anything else is no match.
817fn read_into(read: SynExpr, x: SynIdent, then: SynExpr, at: SynSpan): SynExpr {
818  mat(
819    read,
820    Cons(
821      arm(pctor("Some", Cons(PtBind(x, None), Nil)), then),
822      Cons(arm(pctor("None", Nil), none()), Nil)
823    ),
824    at
825  )
826}
827
828fn seg_pat(ss: List(Seg), n: u64, at: SynSpan): SynPattern × List(SynIdent) {
829  match ss {
830    Nil => #(pctor("Nil", Nil), Nil)
831    Cons(s, rest) => {
832      let id := local(concat("seg", Text::of_u64(n)), at)
833      match seg_pat(rest, inc(n), at) {
834        #(pat, ids) => #(pctor("Cons", Cons(PtBind(id, None), Cons(pat, Nil))), Cons(id, ids))
835      }
836    }
837  }
838}
839
840fn check_segs(
841  ss: List(Seg),
842  ids: List(SynIdent),
843  qs: List(QItem),
844  success: SynExpr,
845  q: SynIdent,
846  at: SynSpan
847): SynExpr ! {macro_error, div} {
848  match #(ss, ids) {
849    #(Nil, Nil) => check_queries(qs, success, q, at)
850    #(Cons(Lit(lit), srest), Cons(id, irest)) => iff(
851      call("Text::eq", Cons(v(id), Cons(ExText(lit, at), Nil)), at),
852      check_segs(srest, irest, qs, success, q, at),
853      none(),
854      at
855    )
856    #(Cons(Param(name, ty), srest), Cons(id, irest)) => read_into(
857      read_as(ty, v(id), at),
858      local(name.text, name.span),
859      check_segs(srest, irest, qs, success, q, at),
860      at
861    )
862    #(Nil, Cons(id, irest)) => fail(at, "router: the path and its binders disagree")
863    #(Cons(seg, srest), Nil) => fail(at, "router: the path and its binders disagree")
864  }
865}
866
867fn check_queries(qs: List(QItem), success: SynExpr, q: SynIdent, at: SynSpan): SynExpr {
868  match qs {
869    Nil => success
870    Cons(item, rest) => {
871      let raw := local(concat(item.name.text, "_raw"), at)
872      let x := local(item.name.text, item.name.span)
873      let found := call("query_value", Cons(v(q), Cons(ExText(item.name.text, at), Nil)), at)
874      let more := check_queries(rest, success, q, at)
875      if item.optional {
876        // Absent is `None`; present and unreadable is no match.
877        let got := local(concat(item.name.text, "_got"), at)
878        mat(
879          found,
880          Cons(
881            arm(
882              pctor("Some", Cons(PtBind(raw, None), Nil)),
883              read_into(
884                read_as(item.ty, v(raw), at),
885                got,
886                let_in(x, some(v(got), at), more, at),
887                at
888              )
889            ),
890            Cons(arm(pctor("None", Nil), let_in(x, none(), more, at)), Nil)
891          ),
892          at
893        )
894      } else {
895        read_into(found, raw, read_into(read_as(item.ty, v(raw), at), x, more, at), at)
896      }
897    }
898  }
899}

901fn try_route(f: Flat, segs: SynIdent, q: SynIdent, at: SynSpan): SynExpr ! {macro_error, div} {
902  if f.catch_all {
903    some(ctor_value(f, at), at)
904  } else {
905    match seg_pat(f.segs, 0, at) {
906      #(pat, ids) => mat(
907        v(segs),
908        Cons(
909          arm(pat, check_segs(f.segs, ids, f.queries, some(ctor_value(f, at), at), q, at)),
910          Cons(arm(PtWild, none()), Nil)
911        ),
912        at
913      )
914    }
915  }
916}
917
918fn decide(
919  fs: List(Flat),
920  segs: SynIdent,
921  q: SynIdent,
922  at: SynSpan,
923  n: u64
924): SynExpr ! {macro_error, div} {
925  match fs {
926    Nil => fail(at, "router: write `/*`")
927    Cons(f, Nil) => ctor_value(f, at)
928    Cons(f, rest) => {
929      let picked := local(concat("picked", Text::of_u64(n)), at)
930      mat(
931        try_route(f, segs, q, at),
932        Cons(
933          arm(pctor("Some", Cons(PtBind(picked, None), Nil)), v(picked)),
934          Cons(arm(pctor("None", Nil), decide(rest, segs, q, at, inc(n))), Nil)
935        ),
936        at
937      )
938    }
939  }
940}
941
942fn any_query(fs: List(Flat)): Bool {
943  match fs {
944    Nil => false
945    Cons(f, rest) => match f.queries {
946      Nil => any_query(rest)
947      Cons(q, more) => true
948    }
949  }
950}
951
952fn route_expr(fs: List(Flat), req: SynIdent, at: SynSpan): SynExpr ! {macro_error, div} {
953  let segs := local("segs", at)
954  let q := local("query", at)
955  let_in(
956    segs,
957    call("segments", Cons(call("HttpRequest::path", Cons(v(req), Nil), at), Nil), at),
958    if any_query(fs) {
959      let_in(q, call("HttpRequest::query", Cons(v(req), Nil), at), decide(fs, segs, q, at, 0), at)
960    } else {
961      decide(fs, segs, q, at, 0)
962    },
963    at
964  )
965}

1005fn emit(
1006  model: SynIdent,
1007  row: List(SynRowEntry),
1008  fs: List(Flat),
1009  at: SynSpan
1010): List(SynStmt) ! {macro_error, div} {
1011  let req := local("request", at)
1012  let route := local("route", at)
1013  let route_ty := v(model)
1014  [
1015    StEnum(SynData {
1016      kind: Some("shared"),
1017      name: model,
1018      generics: Nil,
1019      ctors: List::map((f: Flat) => ctor_of(f, at), fs),
1020      span: at,
1021      unused_generics: Nil
1022    }),
1023    StFn(fn_of(
1024      Syn::decl_name(model, "route"),
1025      [req],
1026      [Some(false)],
1027      [Some(quote { HttpRequest })],
1028      route_ty,
1029      None,
1030      route_expr(fs, req, at)
1031    )),
1032    StFn(fn_of(
1033      Syn::decl_name(model, "handler"),
1034      [req, route],
1035      [None, None],
1036      [Some(quote { HttpRequest }), Some(route_ty)],
1037      quote { Response },
1038      Some(row),
1039      handler_expr(fs, req, route, at)
1040    ))
1041  ]
1042}

1044pub macro fn router(spec: router_syntax::Router): List(SynStmt) ! {macro_error, div} {
1045  let model := Syn::decl_name(Syn::ident(spec.name), spec.name.text)
1046  let fs := lower_all(spec.entry, Nil, middleware_of(spec.entry), model, false)
1047  check_names(fs, Nil)
1048  check_tail(fs)
1049  emit(model, row_of(spec.row), fs, spec.span)
