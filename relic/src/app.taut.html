6// One request in text, the response in text. A request that does not parse is
7// 400. A body that is not UTF-8 has no text form (`response_text` is empty);
8// the socket path writes those bytes unchanged.
9pub fn render<R, !e>(
10  raw: Text,
11  pick: (&HttpRequest) -> R,
12  answer: (HttpRequest, R) -> Response ! e
13): Text ! e {
14  match parse_request(raw) {
15    None => response_text(false, text_response(Status::bad_request(), "malformed request"))
16    Some(r) => {
17      let head := is_head(r)
18      let route := pick(&r)
19      response_text(head, answer(r, route))
20    }
21  }
22}
