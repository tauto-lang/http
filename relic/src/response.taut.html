8// What a handler returns. The body is `Bytes`: shared, so handing it on is a
9// reference count, not a copy of the bytes. Headers are checked before they
10// are written; a name or value that could split the response becomes 500.
11pub struct Response {
12  status: Status,
13  headers: List(@name Text × @value Text),
14  body: Bytes
15}
16
17pub fn text_response(status: Status, body: Text): Response {
18  Response {
19    status: status,
20    headers: [#(@name "content-type", @value "text/plain; charset=utf-8")],
21    body: Bytes::of_text(body)
22  }
23}
24
25pub fn bytes_response(status: Status, body: Bytes): Response {
26  Response {
27    status: status,
28    headers: [#(@name "content-type", @value "application/octet-stream")],
29    body: body
30  }
31}
32
33pub fn with_header(r: Response, name: Text, value: Text): Response {
34  Response {
35    status: r.status,
36    headers: Cons(#(@name name, @value value), r.headers),
37    body: r.body
38  }
39}
40
41// `hs` was consed as headers were set, so the last one is first.
42pub fn prepend_headers(r: Response, hs: List(@name Text × @value Text)): Response {
43  Response {
44    status: r.status,
45    headers: List::append(List::reverse(hs), r.headers),
46    body: r.body
47  }
48}
49
50pub fn method_not_allowed(allow: Text): Response {
51  with_header(text_response(Status::method_not_allowed(), "Method Not Allowed"), "allow", allow)
52}
53
54fn invalid(): Response {
55  text_response(Status::internal_server_error(), "invalid response header")
56}
57
58fn gated(r: Response): Response {
59  if headers_valid_text(r.headers) { r } else { invalid() }
60}
61
62fn header_line(h: @name Text × @value Text): Text {
63  concat((h: @name Text), concat(": ", concat((h: @value Text), "\r\n")))
64}
65
66fn header_lines(hs: List(@name Text × @value Text)): Text {
67  match hs {
68    Nil => ""
69    Cons(h, rest) => concat(header_line(h), header_lines(rest))
70  }
71}
72
73fn status_line(s: Status): Text {
74  concat("HTTP/1.1 ", concat(Text::of_nat(Status::code(s)), concat(" ", concat(reason(s), "\r\n"))))
75}
76
77// A `content-length` or `transfer-encoding` among the handler's headers is dropped (the
78// names compared without ASCII case, as `sent_headers` of `http/src/wire.taut` does): the
79// body goes out by its length, and the one `content-length` written is the body's.
80fn head_text(r: Response): Text {
81  concat(
82    status_line(r.status),
83    concat(
84      header_lines(without_framing(r.headers)),
85      concat("content-length: ", concat(Text::of_u64(Bytes::len(r.body)), "\r\n\r\n"))
86    )
87  )
88}

90// The bytes on the wire. `head` is a HEAD request: the length is the body's,
91// and the body itself is not written.
92pub fn response_bytes(head: Bool, res: Response): Bytes {
93  let safe := gated(res)
94  let prefix := Bytes::of_text(head_text(safe))
95  if head { prefix } else { Bytes::concat(prefix, safe.body) }
96}
97
98// The same bytes, when they are UTF-8. A binary body has no text form.
99pub fn response_text(head: Bool, res: Response): Text {
100  match Text::of_bytes(response_bytes(head, res)) {
101    Some(t) => t
102    None => ""
103  }
104}

106pub fn is_head(r: HttpRequest): Bool {
107  Method::eq(HttpRequest::method(r), Head)
108}

110// Whether a header is called `connection`, whatever the case of its name.
111fn is_connection(h: @name Text × @value Text): Bool {
112  bytes_eq_ci(Text::bytes((h: @name Text)), Text::bytes("connection"))
113}
114
115// The headers without any `connection`.
116fn without_connection(hs: List(@name Text × @value Text)): List(@name Text × @value Text) {
117  match hs {
118    Nil => Nil
119    Cons(h, rest) => keep_unless(is_connection(h), h, without_connection(rest))
120  }
121}
122
123// Whether to keep the connection is relic's to say, not the handler's: any `connection`
124// among the handler's headers is dropped (the name compared without ASCII case).
125// `close` then adds the one `connection: close`, for the last request on a connection;
126// otherwise the response has no `connection` header.
127pub fn finish_response(head: Bool, close: Bool, res: Response): Bytes {
128  let kept := Response {
129    status: res.status,
130    headers: without_connection(res.headers),
131    body: res.body
132  }
133  response_bytes(head, if close { with_header(kept, "connection", "close") } else { kept })
134}
