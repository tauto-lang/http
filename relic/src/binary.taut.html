26// A small HTTP/1.1 server for immutable binary assets. One request per connection.
27// The callback is pure: callers can preload files before opening the listener.
28pub(open) struct(shared) BinaryRequest { method: Text, path: Text }
29pub(open) struct(shared) BinaryResponse { status: u64, content_type: Text, body: Bytes }

31// A count that cannot go on (`overflow`) is answered on the safe side: the
32// header is refused, the request is not complete.
33fn has_line_break(t: Text): Bool ! {div} {
34  handle {
35    var i := 0
36    var found := false
37    while i < Text::len(t) {
38      match Text::byte(t, i) {
39        Some(b) => if U64::of_u8(b) == 10 || U64::of_u8(b) == 13 { found := true } else { #() },
40        None => #()
41      }
42      i := i + 1
43    }
44    found
45  } with { overflow() => true }
46}
47fn complete(t: Text): Bool ! {div} {
48  handle {
49    var i := 0
50    var found := false
51    while i + 3 < Text::len(t) {
52      if slice_text(t, i, i+4) == "\r\n\r\n" { found := true } else { #() }
53      i := i + 1
54    }
55    found
56  } with { overflow() => false }
57}
58pub fn binary_wire(r: BinaryResponse, head: Bool): Bytes ! {div} {
59  let safe := if has_line_break(r.content_type) {
60    BinaryResponse {
61      status: 500,
62      content_type: "text/plain",
63      body: Bytes::of_text("Invalid response header")
64    }
65  } else { r }
66  // The reason phrase of the status (RFC 9110); empty for a status without one, which
67  // HTTP/1.1 allows.
68  let reason := reason_in(Nat::of_u64(safe.status), reasons())
69  let prefix := Bytes::of_text(
70    join_text(
71      [
72        "HTTP/1.1 ",
73        Text::of_u64(safe.status),
74        " ",
75        reason,
76        "\r\ncontent-type: ",
77        safe.content_type,
78        "\r\ncontent-length: ",
79        Text::of_u64(Bytes::len(safe.body)),
80        "\r\nconnection: close\r\nx-content-type-options: nosniff\r\n\r\n"
81      ]
82    )
83  )
84  if head { prefix } else { Bytes::concat(prefix, safe.body) }
85}
86fn failure(status: u64): BinaryResponse {
87  BinaryResponse {
88    status: status,
89    content_type: "text/plain",
90    body: Bytes::of_text("Request rejected")
91  }
92}
93pub fn binary_request(raw: Text): Option(BinaryRequest) ! {div} {
94  // Only the request line is considered; a bounded, complete head is required.
95  if not(complete(raw)) { None } else {
96    handle {
97      var end := 0
98      while end < Text::len(raw) && slice_text(raw, end, end+1) != "\r" { end := end + 1 }
99      match words(slice_text(raw, 0, end)) {
100        Cons(method, Cons(path, Cons(version, Nil))) => if version == "HTTP/1.1" && slice_text(
101          path,
102          0,
103          1
104        ) == "/" { Some(BinaryRequest { method: method, path: path }) } else { None }
105        Nil => None
106        Cons(method, Nil) => None
107        Cons(method, Cons(path, Nil)) => None
108        Cons(method, Cons(path, Cons(version, Cons(extra, more)))) => None
109      }
110    } with { overflow() => None }
111  }
112}
113// When a read or a write that starts now gives up: ten seconds on.
114fn binary_deadline(): i64 ! {monotonic_clock} {
115  let now := perform monotonic_clock()
116  handle { now + 10000000000 } with { overflow() => now }
117}

119// A connection given back with the text read from it.
120struct(affine) Received { conn: TcpStream, raw: Text }
121
122fn receive(c: TcpStream, deadline: i64, raw: Text): Received ! {tcp_read, div} {
123  if Text::len(raw) > 16384 { Received { conn: c, raw: raw } } else {
124    match perform tcp_read(c, 4096, Some(deadline)) {
125      TcpRead { stream, got } => match got {
126        Ok(bytes) => if Bytes::len(bytes) == 0 { Received { conn: stream, raw: raw } } else {
127          match Text::of_bytes(bytes) {
128            Some(t) => {
129              let more := concat(raw, t)
130              if complete(more) { Received { conn: stream, raw: more } } else {
131                receive(stream, deadline, more)
132              }
133            }
134            None => Received { conn: stream, raw: raw }
135          }
136        }
137        Err(_) => Received { conn: stream, raw: raw }
138      }
139    }
140  }
141}

158fn answer<!h: Host>(c: TcpStream, f: (BinaryRequest) -> BinaryResponse ! h): Unit ! {BinaryIo} ∪ h {
159  match receive(c, binary_deadline(), "") {
160    Received { conn, raw } => {
161      let request := binary_request(raw)
162      // The connection waits in a mutex while `f` runs: `f` may resume more than
163      // once, and a connection may not be held across that.
164      let parked := perform mutex_new(conn)
165      let response := if Text::len(raw) > 16384 { failure(431) } else { match request {
166          Some(req) => if req.method == "GET" || req.method == "HEAD" { f(req) } else {
167            failure(405)
168          }
169          None => failure(400)
170        } }
171      let head := match request { Some(req) => req.method == "HEAD", None => false }
172      match perform tcp_write(
173        Mutex::take(parked),
174        binary_wire(response, head),
175        Some(binary_deadline())
176      ) {
177        TcpWritten { stream, result } => {
178          match result { Ok(_) => #(), Err(e) => perform stderr(io_message(e)) }
179          match perform tcp_shutdown(
180            TcpStream::stopper(&stream),
181            ShutdownWrite
182          ) { Ok(_) => #(), Err(_) => #() }
183          perform tcp_close(stream)
184        }
185      }
186    }
187  }
188}
189fn accept_each<!h: Host>(
190  l: TcpListener,
191  f: (BinaryRequest) -> BinaryResponse ! h
192): Unit ! {BinaryIo, tcp_accept} ∪ h {
193  match perform tcp_accept(l, None) {
194    TcpAccepted { listener, got } => match got {
195      Ok(c) => {
196        let parked := perform mutex_new(listener)
197        answer(c, f)
198        accept_each(Mutex::take(parked), f)
199      }
200      // Accepting that is not permitted stops the server; any other failure (out of file
201      // descriptors, say) is reported, and accepting goes on after a short wait.
202      Err(Denied(m)) => perform stderr(concat(m, "\n"))
203      Err(e) => {
204        perform stderr(concat(io_message(e), "\n"))
205        perform sleep(accept_retry_ms())
206        accept_each(listener, f)
207      }
208    }
209  }
210}
211pub fn serve_binary<!h: Host>(
212  address: Text,
213  f: (BinaryRequest) -> BinaryResponse ! h
214): Unit ! {BinaryIo, stop, tcp_accept, tcp_listen} ∪ h {
215  match perform tcp_listen(address) {
216    Err(e) => perform stop(io_message(e))
217    Ok(listener) => {
218      perform stderr(concat("listening on ", TcpListener::address(&listener)))
219      accept_each(listener, f)
220    }
221  }
222}

224fn slice_text(
225  t: Text,
226  a: u64,
227  b: u64
228): Text { match Text::slice(t, a, b) { Some(s) => s, None => "" } }
229fn join_text(
230  xs: List(Text)
231): Text ! {div} { match xs { Nil => "", Cons(x, rest) => concat(x, join_text(rest)) } }
