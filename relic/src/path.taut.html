6// Path segments, with every empty piece dropped: those from a leading or trailing `/`
7// and from a doubled one (`/users//1` is two segments).
8// `/` is no segments. `/healthz` is one. `/users/1` is two.
9// The path is cut at each `/` and the pieces are passed on as they are, not percent-decoded:
10// `/users/a%20b` gives `a%20b`, and `%2F` does not split. (Query values are decoded, below.)
11pub fn segments(path: Text): List(Text) {
12  drop_empty(split_text(path, 47))
13}
14
15fn drop_empty(xs: List(Text)): List(Text) {
16  match xs {
17    Nil => Nil
18    Cons(s, rest) => if s == "" { drop_empty(rest) } else { Cons(s, drop_empty(rest)) }
19  }
20}
21
22// The value of `key` in a query, percent-decoded. `None` when the key is
23// absent. A key with no `=` has the empty value.
24pub fn query_value(q: Option(Text), key: Text): Option(Text) {
25  match q {
26    None => None
27    Some(t) => find_key(split_text(t, 38), key)
28  }
29}
30
31fn find_key(parts: List(Text), key: Text): Option(Text) {
32  match parts {
33    Nil => None
34    Cons(p, rest) => match piece(p) {
35      #(k, v) => if percent_decode(k) == key { Some(percent_decode(v)) } else {
36        find_key(rest, key)
37      }
38    }
39  }
40}
41
42fn piece(p: Text): Text × Text {
43  match split_text(p, 61) {
44    Nil => #("", "")
45    Cons(k, rest) => match rest {
46      Nil => #(k, "")
47      Cons(y, more) => #(k, join_eq(rest))
48    }
49  }
50}
51
52fn join_eq(xs: List(Text)): Text {
53  match xs {
54    Nil => ""
55    Cons(x, rest) => match rest {
56      Nil => x
57      Cons(y, more) => concat(x, concat("=", join_eq(rest)))
58    }
59  }
60}

62// Reading a path segment or a query value as one of the types `router!`
63// knows. Another type `T` is read by `T::from_path(segment: Text): Option(T)`,
64// written next to `T`; a type of another package is wrapped in a newtype.
65
66// Any segment that is not empty.
67pub fn path_text(t: Text): Option(Text) {
68  if t == "" { None } else { Some(t) }
69}
70
71fn digits_of(bs: List(Nat), acc: Nat): Option(Nat) {
72  match bs {
73    Nil => Some(acc)
74    Cons(b, rest) => if b >= 48 && b <= 57 {
75      digits_of(rest, Nat::add(Nat::mul(acc, 10), Nat::sub(b, 48)))
76    } else {
77      None
78    }
79  }
80}

82// Decimal digits only, at least one.
83pub fn path_nat(t: Text): Option(Nat) {
84  match Text::bytes(t) {
85    Nil => None
86    Cons(b, rest) => digits_of(Cons(b, rest), 0)
87  }
88}
89
90// Decimal digits only, at most `u64`'s largest.
91pub fn path_u64(t: Text): Option(u64) {
92  match path_nat(t) {
93    None => None
94    Some(n) => if n <= 18446744073709551615 { Some(U64::of_nat(n)) } else { None }
95  }
96}

98// Decimal digits after an optional `-`, with the digits no greater than `i64`'s largest
99// value, 9223372036854775807. So `-9223372036854775808` is not accepted.
100pub fn path_i64(t: Text): Option(i64) {
101  let magnitude := match Text::bytes(t) {
102    Cons(45, rest) => #(true, match rest {
103      Nil => None
104      Cons(b, more) => digits_of(rest, 0)
105    })
106    Nil => #(false, path_nat(t))
107    Cons(first, others) => #(false, path_nat(t))
108  }
109  match magnitude {
110    #(neg, None) => None
111    #(neg, Some(n)) => if n > 9223372036854775807 {
112      None
113    } else {
114      let x := I64::of_u64(U64::of_nat(n))
115      if neg { handle { Some(0 - x) } with { overflow() => None } } else { Some(x) }
116    }
117  }
