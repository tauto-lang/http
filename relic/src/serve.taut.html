7// Listen, and answer each request with `pick` then `answer`. `pick` is the
8// pure route; `answer` is the effectful handler, whose row `h` is a part of
9// `Host` that `serve` adds to its own: the row is what the server and the
10// handler ask of the host, nothing else. At most `connection_limit()`
11// connections are served at once.
12pub fn serve<R: Shared, !h: Host>(
13  address: Text,
14  pick: (&HttpRequest) -> R,
15  answer: (HttpRequest, R) -> Response ! h
16): Unit ! {ServeIo, ∀ T. spawn(() -> T ! {TaskIo} ∪ h)} ∪ h {
17  serve_response(address, (r: HttpRequest) => {
18    let route := pick(&r)
19    answer(r, route)
20  })
21}
