9// Whether `c` is absent from `l`.
10pub fn no_byte(c: Nat, l: List(Nat)): Bool {
11  match l {
12    Nil => true
13    Cons(h, t) => not(Nat::eq(h, c)) && no_byte(c, t)
14  }
15}

17pub theorem no_byte_cons_and(
18  x: Bool,
19  p: Bool,
20  q: Bool,
21  r: Bool,
22  e: r = (p && q)
23): Bool::and(not(x), r) = (not(x) && p && q) {
24  match x { true => Refl, false => e }
25}

27pub theorem no_byte_append(
28  c: Nat,
29  a: List(Nat),
30  b: List(Nat)
31): no_byte(c, List::append(a, b)) = (no_byte(c, a) && no_byte(c, b)) {
32  match a {
33    Nil => Refl
34    Cons(h, t) => no_byte_cons_and(
35      Nat::eq(h, c),
36      no_byte(c, t),
37      no_byte(c, b),
38      no_byte(c, List::append(t, b)),
39      no_byte_append(c, t, b)
40    )
41  }
42}

44// From `(p && q) = true`, each side.
45pub theorem yes_left(p: Bool, q: Bool, e: Bool::and(p, q) = true): p = true {
46  And::left(Bool::and_split(p, q, e))

53// The first byte of a list without `c` is not `c`.
54pub theorem no_byte_head(
55  c: Nat,
56  h: Nat,
57  t: List(Nat),
58  e: no_byte(c, Cons(h, t)) = true
59): Nat::eq(h, c) = false {
60  Bool::not_true(Nat::eq(h, c), yes_left(not(Nat::eq(h, c)), no_byte(c, t), e))
61}
62
63pub theorem no_byte_tail(
64  c: Nat,
65  h: Nat,
66  t: List(Nat),
67  e: no_byte(c, Cons(h, t)) = true
68): no_byte(c, t) = true {
69  yes_right(not(Nat::eq(h, c)), no_byte(c, t), e)
70}

72// Both halves of an appended list lack `c`.
73pub theorem no_byte_split(
74  c: Nat,
75  a: List(Nat),
76  b: List(Nat),
77  e: no_byte(c, List::append(a, b)) = true
78): no_byte(c, a) = true ∧ no_byte(c, b) = true {
79  Bool::and_split(no_byte(c, a), no_byte(c, b), Eq::trans(Eq::sym(no_byte_append(c, a, b)), e))
80}

88// What follows `p` at the start of `l`, if `l` starts with `p`.
89pub fn strip_prefix(p: List(Nat), l: List(Nat)): Option(List(Nat)) {
90  match p {
91    Nil => Some(l)
92    Cons(a, ps) => ⊢ Option(List(Nat)) from match l {
93      Nil => None
94      Cons(b, t) => ⊢ Option(List(Nat)) from match Nat::eq(a, b) {
95        true => strip_prefix(ps, t)
96        false => None
97      }
98    }
99  }
100}

126pub fn take_until_step(
127  y: Bool,
128  h: Nat,
129  t: List(Nat),
130  rest: @before List(Nat) × @after List(Nat)
131): @before List(Nat) × @after List(Nat) {
132  match y {
133    true => #(@before Nil, @after Cons(h, t))
134    false => cons_fst(h, rest)
135  }

138// The bytes before the first `c`, and the rest from that `c` on.
139pub fn take_until(c: Nat, l: List(Nat)): @before List(Nat) × @after List(Nat) {
140  match l {
141    Nil => #(@before Nil, @after Nil)
142    // `take_until_step(Nat::eq(h, c), h, t, take_until(c, t))`, written out so that a
143    // program stops at the first `c`.
144    Cons(h, t) => ⊢ @before List(Nat) × @after List(Nat) from match Nat::eq(h, c) {
145      true => #(@before Nil, @after Cons(h, t))
146      false => cons_fst(h, take_until(c, t))
147    }
148  }
149}

151pub theorem take_until_append(
152  c: Nat,
153  xs: List(Nat),
154  r: List(Nat),
155  e: no_byte(c, xs) = true
156): take_until(c, List::append(xs, Cons(c, r))) = #(@before xs, @after Cons(c, r)) {
157  match xs {
158    Nil => Eq::congr(⟦take_until_step(□, c, r, take_until(c, r))⟧, Nat::eq_refl(c))
159    Cons(h, t) => Eq::trans(
160      Eq::congr(
161        ⟦take_until_step(
162          □,
163          h,
164          List::append(t, Cons(c, r)),
165          take_until(c, List::append(t, Cons(c, r)))
166        )⟧,
167        no_byte_head(c, h, t, e)
168      ),
169      Eq::congr(⟦cons_fst(h, □)⟧, take_until_append(c, t, r, no_byte_tail(c, h, t, e)))
170    )
171  }
172}

333// Splitting on `c` after a piece without `c` gives that piece first.
334pub theorem split_append(
335  c: Nat,
336  xs: List(Nat),
337  r: List(Nat),
338  e: no_byte(c, xs) = true
339): Bytes::split(List::append(xs, Cons(c, r)), c) = split_pair(xs, Bytes::split(r, c)) {
340  ⊢ Bytes::split(
341    List::append(xs, Cons(c, r)),
342    c
343  ) = split_pair(xs, Bytes::split(r, c)) from match xs {
344    Nil => split_step_sep(c, Bytes::split(r, c))
345    Cons(h, t) => Eq::trans(
346      split_step_no(h, c, Bytes::split(List::append(t, Cons(c, r)), c), no_byte_head(c, h, t, e)),
347      Eq::trans(
348        Eq::congr(⟦split_cons(h, □)⟧, split_append(c, t, r, no_byte_tail(c, h, t, e))),
349        split_cons_pair(h, t, Bytes::split(r, c))
350      )
351    )
352  }
353}

378// Splitting pieces joined with `c` gives them back, when none has `c`.
379pub theorem split_join(
380  c: Nat,
381  p: List(Nat),
382  rest: List(List(Nat)),
383  hp: no_byte(c, p) = true,
384  hr: none_has(c, rest) = true
385): Bytes::split(List::append(p, Bytes::join_rest(Cons(c, Nil), rest)), c) = #(rest, p) {
386  match rest {
387    Nil => Eq::trans(Eq::congr(⟦Bytes::split(□, c)⟧, List::append_nil(p)), split_none(c, p, hp))
388    Cons(q, qs) => Eq::trans(
389      split_append(c, p, List::append(q, Bytes::join_rest(Cons(c, Nil), qs)), hp),
390      Eq::congr(
391        ⟦split_pair(p, □)⟧,
392        split_join(
393          c,
394          q,
395          qs,
396          yes_left(no_byte(c, q), none_has(c, qs), hr),
397          yes_right(no_byte(c, q), none_has(c, qs), hr)
398        )
399      )
400    )
401  }
402}

427pub fn byte_list_eq(a: List(Nat), b: List(Nat)): Bool {
428  match a {
429    Nil => ⊢ Bool from match b {
430      Nil => true
431      Cons(y, u) => false
432    }
433    Cons(x, t) => ⊢ Bool from match b {
434      Nil => false
435      Cons(y, u) => Nat::eq(x, y) && byte_list_eq(t, u)
436    }
437  }
438}

440pub theorem bytes_eq_refl(a: List(Nat)): byte_list_eq(a, a) = true {
441  match a {
442    Nil => Refl
443    Cons(x, t) => Eq::trans(Eq::congr(⟦□ && byte_list_eq(t, t)⟧, Nat::eq_refl(x)), bytes_eq_refl(t))
444  }
445}

449  match Nat::le(65, b) && Nat::le(b, 90) {

469// The bytes of `t` from `start` to `end`. The splitting below cuts only next to a byte
470// that is ASCII (a separator such as `/`, `&`, `=` or a space), so the pieces of UTF-8 text
471// are UTF-8 text again. Cutting next to a byte of 128 or more could split a character.
472pub fn cut(t: Text, start: Nat, end: Nat): Text {
473  Text::Mk(List::take(Nat::sub(end, start), List::drop(start, Text::bytes(t))))
474}

514// What is left of `bs` from `i` on, read by its first element.
515pub fn drop_at(bs: List(Nat), i: Nat): List(Nat) {
516  match List::nth(bs, i) {
517    None => Nil
518    Some(x) => Cons(x, List::drop(Succ(i), bs))
519  }
520}
521
522pub theorem drop_nth(bs: List(Nat), i: Nat): List::drop(i, bs) = drop_at(bs, i) {
523  match bs {
524    Nil => ⊢ List::drop(i, Nil(<A: Nat>)) = drop_at(Nil(<A: Nat>), i) from match i {
525      Zero => Refl
526      Succ(j) => Refl
527    }
528    Cons(h, t) => ⊢ List::drop(i, Cons(h, t)) = drop_at(Cons(h, t), i) from match i {
529      Zero => Refl
530      Succ(j) => drop_nth(t, j)
531    }
532  }
533}

535// `bytes_eq_ci` of the bytes of `t` from `i` on and `name`, reading `t` with
536// `Text::byte`: a program builds no list of the text's bytes.
537pub fn text_eq_ci_at(t: Text, i: Nat, name: List(Nat)): Bool {
538  match name {
539    Nil => ⊢ Bool from match Text::byte(t, i) {
540      None => true
541      Some(x) => false
542    }
543    Cons(y, u) => ⊢ Bool from match Text::byte(t, i) {
544      None => false
545      Some(x) => Nat::eq(lower_byte(x), lower_byte(y)) && text_eq_ci_at(t, Succ(i), u)
546    }
547  }
548}

593pub theorem text_eq_ci_on(
594  t: Text,
595  i: Nat,
596  name: List(Nat)
597): text_eq_ci_at(t, i, name) = byte_list_eq(lower(drop_at(Text::bytes(t), i)), lower(name)) {
598  match name {
599    Nil => text_eq_ci_nil(List::nth(Text::bytes(t), i), List::drop(Succ(i), Text::bytes(t)))
600    Cons(y, u) => text_eq_ci_cons(
601      List::nth(Text::bytes(t), i),
602      List::drop(Succ(i), Text::bytes(t)),
603      y,
604      u,
605      text_eq_ci_at(t, Succ(i), u),
606      Eq::trans(
607        text_eq_ci_on(t, Succ(i), u),
608        Eq::congr(⟦byte_list_eq(lower(□), lower(u))⟧, Eq::sym(drop_nth(Text::bytes(t), Succ(i))))
609      )
610    )
611  }
612}
