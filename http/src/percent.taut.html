8// Percent-encoding of URLs and form bodies, on top of the url tote: a byte that RFC 3986
9// does not leave as it is becomes `%XX`, and when reading, `+` is a space as in a form.
10// Decoding what was encoded gives the text back.
11
12fn utf8_or(o: Option(Text), t: Text): Text {
13  match o {
14    None => t
15    Some(x) => x
16  }
17}
18
19// `+` as a space, as in a form's query.
20fn plus_space(l: List(Nat)): List(Nat) {
21  match l {
22    Nil => Nil
23    Cons(b, t) => Cons(⊢ Nat from match Nat::eq(b, 43) {
24      true => 32
25      false => b
26    }, plus_space(t))
27  }
28}

30pub module Percent {
31  pub fn bytes(l: List(Nat)): List(Nat) {
32    pct_decode(plus_space(l))
33  }
34
35  theorem bytes_encode(l: List(Nat), ok: bytes_ok(l) = true): Percent::bytes(encode_bytes(l)) = l {
36    Eq::trans(
37      Eq::congr(
38        ⟦pct_decode(□)⟧,
39        plus_space_id(
40          encode_bytes(l),
41          encode_no_byte(
42            43,
43            l,
44            (b: Nat, u: unreserved(b) = true) => unreserved_plus(b, u),
45            Refl,
46            hex_ne_plus
47          )
48        )
49      ),
50      pct_decode_encode(reserved, Refl, l, ok)
51    )
52  }
53
54  // A UTF-8 text survives `percent_encode` and then `percent_decode`.
55  pub theorem decode_encode(
56    t: Text,
57    ok: bytes_ok(Text::bytes(t)) = true,
58    utf8: utf8_valid(Text::bytes(t)) = true
59  ): percent_decode(percent_encode(t)) = t {
60    match t {
61      Text::Mk(bs) => Eq::trans(
62        Eq::congr(
63          ⟦utf8_or(Text::of_utf8(□), percent_encode(Text::Mk(bs)))⟧,
64          Percent::bytes_encode(bs, ok)
65        ),
66        Eq::congr(⟦utf8_or(text_gate(□, bs), percent_encode(Text::Mk(bs)))⟧, utf8)
67      )
68    }
69  }
70}

72// `%XX` as its byte and `+` as a space. A `%` without two hex digits after it stays as it
73// is. When the bytes are not UTF-8, the text stays undecoded.
74pub fn percent_decode(t: Text): Text {
75  utf8_or(Text::of_utf8(Percent::bytes(Text::bytes(t))), t)
76}

80fn in_range(lo: Nat, hi: Nat, b: Nat): Bool {
81  Nat::le(lo, b) && Nat::le(b, hi)
82}
83
84// The bytes RFC 3986 leaves as they are: letters, digits and `-` `.` `_` `~`.
85// (`+`, `%`, `&` and `=` are not among them; `unreserved` rules them out one by one,
86// and the lemmas below rely on that.)
87fn unreserved_class(b: Nat): Bool {
88  Bool::or(
89    in_range(65, 90, b) || in_range(97, 122, b),
90    Bool::or(
91      in_range(48, 57, b),
92      Nat::eq(b, 45) || Nat::eq(b, 46) || (Nat::eq(b, 95) || Nat::eq(b, 126))
93    )
94  )
95}
96
97fn unreserved(b: Nat): Bool {
98  Bool::and(
99    not(Nat::eq(b, 43)),
100    Bool::and(
101      not(Nat::eq(b, 37)),
102      not(Nat::eq(b, 38)) && (not(Nat::eq(b, 61)) && unreserved_class(b))
103    )
104  )
105}
106
107// The bytes written as `%XX`.
108fn reserved(b: Nat): Bool {
109  not(unreserved(b))
110}
111
112pub fn encode_bytes(l: List(Nat)): List(Nat) {
113  pct_encode(reserved, l)
114}
115
116// Every byte but the unreserved ones as `%XX`, so the text can stand anywhere
117// in a query, a path segment or a form.
118pub fn percent_encode(t: Text): Text {
119  Text::Mk(encode_bytes(Text::bytes(t)))
120}

124fn unreserved_rest(b: Nat): Bool {
125  not(Nat::eq(b, 37)) && unreserved_more(b)
126}
127
128fn unreserved_more(b: Nat): Bool {
129  not(Nat::eq(b, 38)) && (not(Nat::eq(b, 61)) && unreserved_class(b))
130}
131
132theorem unreserved_plus(b: Nat, u: unreserved(b) = true): Nat::eq(b, 43) = false {
133  Bool::not_true(Nat::eq(b, 43), yes_left(not(Nat::eq(b, 43)), unreserved_rest(b), u))
134}
135
136theorem unreserved_tail(b: Nat, u: unreserved(b) = true): unreserved_rest(b) = true {
137  yes_right(not(Nat::eq(b, 43)), unreserved_rest(b), u)
138}
139
140theorem unreserved_more_of(b: Nat, u: unreserved(b) = true): unreserved_more(b) = true {
141  yes_right(not(Nat::eq(b, 37)), unreserved_more(b), unreserved_tail(b, u))
142}
143
144theorem unreserved_amp(b: Nat, u: unreserved(b) = true): Nat::eq(b, 38) = false {
145  Bool::not_true(
146    Nat::eq(b, 38),
147    yes_left(
148      not(Nat::eq(b, 38)),
149      not(Nat::eq(b, 61)) && unreserved_class(b),
150      unreserved_more_of(b, u)
151    )
152  )
153}
154
155theorem unreserved_equals(b: Nat, u: unreserved(b) = true): Nat::eq(b, 61) = false {
156  Bool::not_true(
157    Nat::eq(b, 61),
158    yes_left(
159      not(Nat::eq(b, 61)),
160      unreserved_class(b),
161      yes_right(
162        not(Nat::eq(b, 38)),
163        not(Nat::eq(b, 61)) && unreserved_class(b),
164        unreserved_more_of(b, u)
165      )
166    )
167  )
168}

170theorem not_false(x: Bool, e: not(x) = false): x = true {
171  match x {
172    true => Refl
173    false => False::elim(true_ne_false(e))
174  }

177// A hex digit is never `+`, `&` or `=`.
178theorem hex_ne_plus(d: Nat): Nat::eq(hex_digit(d), 43) = false {
179  ⊢ Nat::eq(
180    ⊢ Nat from match Nat::lt(d, 10) { true => Nat::add(d, 48), false => Nat::add(d, 55) },
181    43
182  ) = false from match Nat::lt(d, 10) eqn e {
183    false => Refl
184    true => Refl
185  }
186}
187
188theorem hex_ne_amp(d: Nat): Nat::eq(hex_digit(d), 38) = false {
189  ⊢ Nat::eq(
190    ⊢ Nat from match Nat::lt(d, 10) { true => Nat::add(d, 48), false => Nat::add(d, 55) },
191    38
192  ) = false from match Nat::lt(d, 10) eqn e {
193    false => Refl
194    true => Refl
195  }
196}
197
198theorem hex_ne_equals(d: Nat): Nat::eq(hex_digit(d), 61) = false {
199  ⊢ Nat::eq(
200    ⊢ Nat from match Nat::lt(d, 10) { true => Nat::add(d, 48), false => Nat::add(d, 55) },
201    61
202  ) = false from match Nat::lt(d, 10) eqn e {
203    false => ⊢ Nat::eq(d, 6) = false from match Nat::eq(d, 6) eqn h {
204      false => Refl
205      true => False::elim(
206        false_ne_true(Eq::trans(Eq::sym(e), Eq::congr(⟦Nat::lt(□, 10)⟧, Nat::eq_sound(d, 6, h))))
207      )
208    }
209    true => ⊢ Nat::eq(d, 13) = false from match Nat::eq(d, 13) eqn h {
210      false => Refl
211      true => False::elim(
212        false_ne_true(Eq::trans(Eq::sym(Eq::congr(⟦Nat::lt(□, 10)⟧, Nat::eq_sound(d, 13, h))), e))
213      )
214    }
215  }

218// The encoding writes no `c` when `c` is not unreserved, not `%` and not a hex digit.
219theorem encode_no_byte(
220  c: Nat,
221  l: List(Nat),
222  hu: (b: Nat, u: unreserved(b) = true) -> Nat::eq(b, c) = false,
223  h37: Nat::eq(37, c) = false,
224  hhex: (d: Nat) -> Nat::eq(hex_digit(d), c) = false
225): no_byte(c, encode_bytes(l)) = true {
226  match l {
227    Nil => Refl
228    Cons(b, t) => ⊢ no_byte(
229      c,
230      ⊢ List(Nat) from match reserved(b) {
231        true => Cons(37, Cons(hex_digit(Nat::rem_nz(Nat::div_nz(b, 16, Nat::ne_zero(16, Refl)), 16, Nat::ne_zero(16, Refl))), Cons(hex_digit(Nat::rem_nz(b, 16, Nat::ne_zero(16, Refl))), encode_bytes(t))))
232        false => Cons(b, encode_bytes(t))
233      }
234    ) = true from match reserved(b) eqn e {
235      false => Eq::trans(
236        Eq::congr(⟦not(□) && no_byte(c, encode_bytes(t))⟧, hu(b, not_false(unreserved(b), e))),
237        encode_no_byte(c, t, hu, h37, hhex)
238      )
239      true => Eq::trans(
240        Eq::congr(
241          ⟦not(□) && (
242            not(
243              Nat::eq(
244                hex_digit(
245                  Nat::rem_nz(
246                    Nat::div_nz(b, 16, Nat::ne_zero(16, Refl)),
247                    16,
248                    Nat::ne_zero(16, Refl)
249                  )
250                ),
251                c
252              )
253            ) && (
254              not(
255                Nat::eq(hex_digit(Nat::rem_nz(b, 16, Nat::ne_zero(16, Refl))), c)
256              ) && no_byte(c, encode_bytes(t))
257            )
258          )⟧,
259          h37
260        ),
261        Eq::trans(
262          Eq::congr(
263            ⟦not(□) && (
264              not(
265                Nat::eq(hex_digit(Nat::rem_nz(b, 16, Nat::ne_zero(16, Refl))), c)
266              ) && no_byte(c, encode_bytes(t))
267            )⟧,
268            hhex(
269              Nat::rem_nz(Nat::div_nz(b, 16, Nat::ne_zero(16, Refl)), 16, Nat::ne_zero(16, Refl))
270            )
271          ),
272          Eq::trans(
273            Eq::congr(
274              ⟦not(□) && no_byte(c, encode_bytes(t))⟧,
275              hhex(Nat::rem_nz(b, 16, Nat::ne_zero(16, Refl)))
276            ),
277            encode_no_byte(c, t, hu, h37, hhex)
278          )
279        )
280      )
281    }
282  }
283}

301// Bytes, as an encoder sees them: each below 256.
302pub fn bytes_ok(l: List(Nat)): Bool {
303  all_bytes(is_byte, l)
304}
305
306// Without `+`, reading `+` as a space changes nothing.
307theorem plus_space_id(l: List(Nat), h: no_byte(43, l) = true): plus_space(l) = l {
308  match l {
309    Nil => Refl
310    Cons(b, t) => Eq::trans(
311      Eq::congr(
312        (x: Bool) => Cons(⊢ Nat from match x { true => 32, false => b }, plus_space(t)),
313        no_byte_head(43, b, t, h)
314      ),
315      Eq::congr(⟦Cons(b, □)⟧, plus_space_id(t, no_byte_tail(43, b, t, h)))
316    )
317  }
318}

320// `Text::of_utf8` once it has checked the bytes.
321fn text_gate(y: Bool, bs: List(Nat)): Option(Text) {
322  match y {
323    true => Some(Text::Mk(bs))
324    false => None
325  }
