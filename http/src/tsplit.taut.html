9// Splitting a text on a byte, reading it where it is: `Text::byte` finds each
10// separator and `Text::slice` cuts each piece, as `Text::split_on` says.
11
12pub fn nosep(sep: Nat, p: List(Nat)): Bool {
13  match p {
14    Nil => true
15    Cons(h, t) => ⊢ Bool from match h == sep {
16      true => false
17      false => nosep(sep, t)
18    }
19  }
20}

22// The bytes of `t` from `s` up to `e`.
23pub fn cut_text(t: Text, s: Nat, e: Nat): Text {
24  match Text::slice(t, s, e) {
25    None => Text::Mk(between(Text::bytes(t), s, e))
26    Some(x) => x
27  }
28}

30// The pieces of `t` from `s`, the current one begun at `s` and read up to `i`.
31pub fn split_from(t: Text, sep: Nat, s: Nat, i: Nat, k: Nat): List(Text) {
32  match k {
33    Zero => Cons(cut_text(t, s, i), Nil)
34    Succ(k2) => ⊢ List(Text) from match Text::byte(t, i) {
35      None => Cons(cut_text(t, s, i), Nil)
36      Some(b) => ⊢ List(Text) from match b == sep {
37        true => Cons(cut_text(t, s, i), split_from(t, sep, Succ(i), Succ(i), k2))
38        false => split_from(t, sep, s, Succ(i), k2)
39      }
40    }
41  }
42}
43
44// `Text::split_on(t, sep)`, read where the bytes are.
45pub fn split_text(t: Text, sep: Nat): List(Text) {
46  split_from(t, sep, 0, 0, Text::len(t))
47}

51pub fn prep(p: List(Nat), q: List(List(Nat)) × List(Nat)): List(List(Nat)) × List(Nat) {
52  match q {
53    #(a, c) => #(a, List::append(p, c))
54  }
55}
56
57pub theorem prep_nil(q: List(List(Nat)) × List(Nat)): prep(Nil, q) = q {
58  match q {
59    #(a, c) => Refl
60  }
61}
62
63pub theorem prep_step(
64  h: Nat,
65  sep: Nat,
66  p: List(Nat),
67  q: List(List(Nat)) × List(Nat),
68  hh: (h == sep) = false
69): Bytes::split_step(h, sep, prep(p, q)) = prep(Cons(h, p), q) {
70  match q {
71    #(a, c) => Eq::congr(
72      (y: Bool) => (⊢ List(List(Nat)) × List(Nat) from match y {
73        true => #(Cons(List::append(p, c), a), List::Nil(<A: Nat>))
74        false => #(a, Cons(h, List::append(p, c)))
75      }),
76      hh
77    )
78  }
79}

81// A piece without the separator goes in front of the first part.
82pub theorem split_prefix(
83  sep: Nat,
84  p: List(Nat),
85  r: List(Nat),
86  hn: nosep(sep, p) = true
87): Bytes::split(List::append(p, r), sep) = prep(p, Bytes::split(r, sep)) {
88  match p {
89    Nil => Eq::sym(prep_nil(Bytes::split(r, sep)))
90    Cons(h, t) => ((hn1: (⊢ Bool from match h == sep { true => false, false => nosep(sep, t) }) = true) =>
91      ⊢ Bytes::split(List::append(Cons(h, t), r), sep) = prep(Cons(h, t), Bytes::split(r, sep)) from match h == sep eqn ye {
92        false => Eq::trans(
93          Eq::congr(⟦Bytes::split_step(h, sep, □)⟧, split_prefix(sep, t, r, hn1)),
94          prep_step(h, sep, t, Bytes::split(r, sep), ye)
95        )
96        true => False::elim(false_ne_true(hn1))
97      })(hn)
98  }
99}

101pub theorem nosep_snoc(
102  sep: Nat,
103  p: List(Nat),
104  x: Nat,
105  hn: nosep(sep, p) = true,
106  hx: (x == sep) = false
107): nosep(sep, List::append(p, Cons(x, Nil))) = true {
108  match p {
109    Nil => Eq::congr((y: Bool) => (⊢ Bool from match y { true => false, false => true }), hx)
110    Cons(h, t) => ((hn1: (⊢ Bool from match h == sep { true => false, false => nosep(sep, t) }) = true) =>
111      ⊢ (⊢ Bool from match h == sep {
112        true => false
113        false => nosep(sep, List::append(t, Cons(x, Nil)))
114      }) = true from match h == sep eqn ye {
115        false => nosep_snoc(sep, t, x, hn1, hx)
116        true => False::elim(false_ne_true(hn1))
117      })(hn)
118  }
119}

121pub theorem drop_some<A: Type>(
122  l: List(A),
123  i: Nat,
124  x: A,
125  ho: List::nth(l, i) = Some(x)
126): List::drop(i, l) = Cons(x, List::drop(Succ(i), l)) {
127  match l {
128    Nil => False::elim(
129      false_ne_true(
130        Eq::congr((o: Option(A)) => (⊢ Bool from match o { None => false, Some(y) => true }), ho)
131      )
132    )
133    Cons(h, t) => ⊢ List::drop(
134      i,
135      Cons(h, t)
136    ) = Cons(x, List::drop(Succ(i), Cons(h, t))) from match i {
137      Zero => Eq::congr(
138        ⟦Cons(□, t)⟧,
139        Eq::congr((o: Option(A)) => (⊢ A from match o { None => h, Some(y) => y }), ho)
140      )
141      Succ(j) => drop_some(t, j, x, ho)
142    }
143  }
144}

146pub theorem drop_none<A: Type>(
147  l: List(A),
148  i: Nat,
149  ho: List::nth(l, i) = None
150): List::drop(i, l) = Nil {
151  match l {
152    Nil => ⊢ List::drop(i, List::Nil(<A: A>)) = Nil from match i {
153      Zero => Refl
154      Succ(j) => Refl
155    }
156    Cons(h, t) => ⊢ List::drop(i, Cons(h, t)) = Nil from match i {
157      Zero => False::elim(
158        false_ne_true(
159          Eq::congr((o: Option(A)) => (⊢ Bool from match o { None => true, Some(y) => false }), ho)
160        )
161      )
162      Succ(j) => drop_none(t, j, ho)
163    }
164  }
165}

169    Nil => Refl

181pub theorem cut_eq(
182  t: Text,
183  s: Nat,
184  e: Nat
185): cut_text(t, s, e) = Text::Mk(between(Text::bytes(t), s, e)) {
186  ⊢ Eq(
187    Text,
188    cut_of(
189      ⊢ Option(Text) from match s <= e && (Text::boundary(t, s) && Text::boundary(t, e)) {
190        false => None
191        true => Some(Text::Mk(List::take(e - s, List::drop(s, Text::bytes(t)))))
192      },
193      Text::Mk(between(Text::bytes(t), s, e))
194    ),
195    Text::Mk(between(Text::bytes(t), s, e))
196  ) from match s <= e && (Text::boundary(t, s) && Text::boundary(t, e)) eqn ge {
197    false => Refl
198    true => Eq::congr((m: List(Nat)) => Text::Mk(m), take_between(Text::bytes(t), s, e))
199  }
200}

202pub theorem end_eq(
203  t: Text,
204  sep: Nat,
205  s: Nat,
206  i: Nat,
207  hn: nosep(sep, between(Text::bytes(t), s, i)) = true,
208  hd: List::drop(i, Text::bytes(t)) = Nil
209): Cons(
210  cut_text(t, s, i),
211  Nil
212) = Text::of_split(
213  Bytes::split(List::append(between(Text::bytes(t), s, i), List::drop(i, Text::bytes(t))), sep)
214) {
215  Eq::sym(
216    Eq::trans(
217      Eq::congr(
218        ⟦Text::of_split(Bytes::split(List::append(between(Text::bytes(t), s, i), □), sep))⟧,
219        hd
220      ),
221      Eq::trans(
222        Eq::congr(⟦Text::of_split(□)⟧, split_prefix(sep, between(Text::bytes(t), s, i), Nil, hn)),
223        Eq::trans(
224          Eq::congr(
225            ⟦Cons(Text::Mk(□), List::Nil(<A: Text>))⟧,
226            List::append_nil(between(Text::bytes(t), s, i))
227          ),
228          Eq::sym(Eq::congr(⟦Cons(□, List::Nil(<A: Text>))⟧, cut_eq(t, s, i)))
229        )
230      )
231    )
232  )
233}

235pub theorem sep_step(
236  p: List(Nat),
237  b: Nat,
238  sep: Nat,
239  q: List(List(Nat)) × List(Nat),
240  hb: (b == sep) = true
241): Text::of_split(prep(p, Bytes::split_step(b, sep, q))) = Cons(Text::Mk(p), Text::of_split(q)) {
242  match q {
243    #(a, c) => Eq::trans(
244      Eq::congr(
245        (y: Bool) => Text::of_split(prep(p, ⊢ List(List(Nat)) × List(Nat) from match y {
246          true => #(Cons(c, a), List::Nil(<A: Nat>))
247          false => #(a, Cons(b, c))
248        })),
249        hb
250      ),
251      Eq::congr(⟦Cons(Text::Mk(□), Text::of_bytes_list(Cons(c, a)))⟧, List::append_nil(p))
252    )
253  }
254}

256pub theorem split_eq(
257  t: Text,
258  sep: Nat,
259  s: Nat,
260  i: Nat,
261  k: Nat,
262  hs: (s <= i) = true,
263  hn: nosep(sep, between(Text::bytes(t), s, i)) = true,
264  hk: (List::length(List::drop(i, Text::bytes(t))) <= k) = true
265): split_from(
266  t,
267  sep,
268  s,
269  i,
270  k
271) = Text::of_split(
272  Bytes::split(List::append(between(Text::bytes(t), s, i), List::drop(i, Text::bytes(t))), sep)
273) {
274  ⊢ split_from(
275    t,
276    sep,
277    s,
278    i,
279    k
280  ) = Text::of_split(
281    Bytes::split(List::append(between(Text::bytes(t), s, i), List::drop(i, Text::bytes(t))), sep)
282  ) from match k {
283    Zero => end_eq(t, sep, s, i, hn, nil_of_len(List::drop(i, Text::bytes(t)), hk))
284    Succ(k2) => opt_cases(
285      (o: Option(Nat)) => Eq(
286        List(Text),
287        ⊢ List(Text) from match o {
288          None => Cons(cut_text(t, s, i), Nil)
289          Some(b) => ⊢ List(Text) from match b == sep {
290            true => Cons(cut_text(t, s, i), split_from(t, sep, Succ(i), Succ(i), k2))
291            false => split_from(t, sep, s, Succ(i), k2)
292          }
293        },
294        Text::of_split(Bytes::split(List::append(between(Text::bytes(t), s, i), List::drop(i, Text::bytes(t))), sep))
295      ),
296      Text::byte(t, i),
297      (oe: Text::byte(t, i) = None) => end_eq(t, sep, s, i, hn, drop_none(Text::bytes(t), i, oe)),
298      (b: Nat) => (oe: Text::byte(t, i) = Some(b)) => Eq::trans(
299        ⊢ Eq(
300          List(Text),
301          ⊢ List(Text) from match b == sep {
302            true => Cons(cut_text(t, s, i), split_from(t, sep, Succ(i), Succ(i), k2))
303            false => split_from(t, sep, s, Succ(i), k2)
304          },
305          Text::of_split(Bytes::split(List::append(between(Text::bytes(t), s, i), Cons(b, List::drop(Succ(i), Text::bytes(t)))), sep))
306        ) from match b == sep eqn ye {
307          false => Eq::trans(
308            split_eq(
309              t, sep, s, Succ(i), k2,
310              Nat::le_trans(s, i, Succ(i), hs, Nat::le_succ(i)),
311              Eq::trans(
312                Eq::congr(⟦nosep(sep, □)⟧, between_snoc(Text::bytes(t), s, i, b, oe, hs)),
313                nosep_snoc(sep, between(Text::bytes(t), s, i), b, hn, ye)
314              ),
315              Eq::trans(
316                Eq::sym(Eq::congr(⟦List::length(□) <= Succ(k2)⟧, drop_some(Text::bytes(t), i, b, oe))),
317                hk
318              )
319            ),
320            Eq::trans(
321              Eq::congr(
322                ⟦Text::of_split(Bytes::split(List::append(□, List::drop(Succ(i), Text::bytes(t))), sep))⟧,
323                between_snoc(Text::bytes(t), s, i, b, oe, hs)
324              ),
325              Eq::congr(
326                ⟦Text::of_split(Bytes::split(□, sep))⟧,
327                List::append_assoc(between(Text::bytes(t), s, i), Cons(b, Nil), List::drop(Succ(i), Text::bytes(t)))
328              )
329            )
330          )
331          true => Eq::trans(
332            Eq::congr(⟦Cons(□, split_from(t, sep, Succ(i), Succ(i), k2))⟧, cut_eq(t, s, i)),
333            Eq::trans(
334              Eq::congr(
335                ⟦Cons(Text::Mk(between(Text::bytes(t), s, i)), □)⟧,
336                Eq::trans(
337                  split_eq(
338                    t, sep, Succ(i), Succ(i), k2,
339                    le_self(Succ(i)),
340                    Eq::congr(⟦nosep(sep, □)⟧, between_same(Text::bytes(t), Succ(i))),
341                    Eq::trans(
342                      Eq::sym(Eq::congr(⟦List::length(□) <= Succ(k2)⟧, drop_some(Text::bytes(t), i, b, oe))),
343                      hk
344                    )
345                  ),
346                  Eq::congr(
347                    ⟦Text::of_split(Bytes::split(List::append(□, List::drop(Succ(i), Text::bytes(t))), sep))⟧,
348                    between_same(Text::bytes(t), Succ(i))
349                  )
350                )
351              ),
352              Eq::sym(
353                Eq::trans(
354                  Eq::congr(
355                    ⟦Text::of_split(□)⟧,
356                    split_prefix(sep, between(Text::bytes(t), s, i), Cons(b, List::drop(Succ(i), Text::bytes(t))), hn)
357                  ),
358                  sep_step(between(Text::bytes(t), s, i), b, sep, Bytes::split(List::drop(Succ(i), Text::bytes(t)), sep), ye)
359                )
360              )
361            )
362          )
363        },
364        Eq::sym(
365          Eq::congr(
366            ⟦Text::of_split(Bytes::split(List::append(between(Text::bytes(t), s, i), □), sep))⟧,
367            drop_some(Text::bytes(t), i, b, oe)
368          )
369        )
370      )
371    )
372  }
373}

375pub theorem split_text_eq(t: Text, sep: Nat): split_text(t, sep) = Text::split_on(t, sep) {
376  Eq::trans(
377    split_eq(
378      t,
379      sep,
380      0,
381      0,
382      Text::len(t),
383      Refl,
384      Eq::congr(⟦nosep(sep, □)⟧, between_same(Text::bytes(t), 0)),
385      le_self(List::length(Text::bytes(t)))
386    ),
387    Eq::congr(
388      ⟦Text::of_split(Bytes::split(List::append(□, Text::bytes(t)), sep))⟧,
389      between_same(Text::bytes(t), 0)
390    )
391  )
