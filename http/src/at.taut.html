44// The front of `between`: the item at `s`, when `s` is before `e` and there is one.
45pub theorem between_front<A: Type>(
46  l: List(A),
47  s: Nat,
48  e: Nat
49): between(l, s, e) = (⊢ List(A) from match s < e {
50  false => Nil
51  true => ⊢ List(A) from match List::nth(l, s) {
52    None => Nil
53    Some(x) => Cons(x, between(l, Succ(s), e))
54  }
55}) {
56  ⊢ between(l, s, e) = (⊢ List(A) from match s < e {
57    false => Nil
58    true => ⊢ List(A) from match List::nth(l, s) {
59      None => Nil
60      Some(x) => Cons(x, between(l, Succ(s), e))
61    }
62  }) from match l {
63    Nil => ⊢ Eq(List(A), List::Nil(<A: A>), ⊢ List(A) from match s < e {
64      false => Nil
65      true => Nil
66    }) from match s < e eqn e2 {
67      false => Refl
68      true => Refl
69    }
70    Cons(h, t) => ⊢ between(Cons(h, t), s, e) = (⊢ List(A) from match s < e {
71      false => Nil
72      true => ⊢ List(A) from match List::nth(Cons(h, t), s) {
73        None => Nil
74        Some(x) => Cons(x, between(Cons(h, t), Succ(s), e))
75      }
76    }) from match e {
77      Zero => Refl
78      Succ(e2) => ⊢ between(Cons(h, t), s, Succ(e2)) = (⊢ List(A) from match s < Succ(e2) {
79        false => Nil
80        true => ⊢ List(A) from match List::nth(Cons(h, t), s) {
81          None => Nil
82          Some(x) => Cons(x, between(Cons(h, t), Succ(s), Succ(e2)))
83        }
84      }) from match s {
85        Zero => Refl
86        Succ(s2) => between_front(t, s2, e2)
87      }
88    }
89  }
