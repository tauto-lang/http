3// A status code: three digits, the first from 1 to 9 (RFC 9110 section 15). A code outside
4// 100 to 999 cannot be written, so every status prints as three digits and reads back as itself.

58    match Nat::eq(b, 48) {
59      true => Some(D0)
60      false => ⊢ Option(Digit) from match Nat::eq(b, 49) {
61        true => Some(D1)
62        false => ⊢ Option(Digit) from match Nat::eq(b, 50) {
63          true => Some(D2)
64          false => ⊢ Option(Digit) from match Nat::eq(b, 51) {
65            true => Some(D3)
66            false => ⊢ Option(Digit) from match Nat::eq(b, 52) {
67              true => Some(D4)
68              false => ⊢ Option(Digit) from match Nat::eq(b, 53) {
69                true => Some(D5)
70                false => ⊢ Option(Digit) from match Nat::eq(b, 54) {
71                  true => Some(D6)
72                  false => ⊢ Option(Digit) from match Nat::eq(b, 55) {
73                    true => Some(D7)
74                    false => ⊢ Option(Digit) from match Nat::eq(b, 56) {
75                      true => Some(D8)
76                      false => ⊢ Option(Digit) from match Nat::eq(b, 57) {

195pub module Status {
196  pub fn code(s: Status): Nat {
197    match s {
198      Mk(l, t, o) => Nat::add(
199        Nat::add(Nat::mul(100, Lead::value(l)), Nat::mul(10, Digit::value(t))),
200        Digit::value(o)
201      )
202    }
203  }
204
205  // The three ASCII digits of the status.
206  pub fn bytes(s: Status): List(Nat) {
207    match s {
208      Mk(l, t, o) => Cons(Lead::byte(l), Cons(Digit::byte(t), Cons(Digit::byte(o), Nil)))
209    }
210  }
211
212  pub fn of_parts(l: Option(Lead), t: Option(Digit), o: Option(Digit)): Option(Status) {
213    match l {
214      None => None
215      Some(x) => ⊢ Option(Status) from match t {
216        None => None
217        Some(y) => ⊢ Option(Status) from match o {
218          None => None
219          Some(z) => Some(Status::Mk(x, y, z))
220        }
221      }
222    }
223  }
224
225  // The status written by three ASCII bytes, if they are three digits and the first is not 0.
226  pub fn of_bytes(a: Nat, b: Nat, c: Nat): Option(Status) {
227    Status::of_parts(Lead::of_byte(a), Digit::of_byte(b), Digit::of_byte(c))
228  }
229
230  // The status at the front of `l`: its first three bytes.
231  pub fn read(l: List(Nat)): Option(Status) {
232    match l {
233      Nil => None
234      Cons(a, r1) => ⊢ Option(Status) from match r1 {
235        Nil => None
236        Cons(b, r2) => ⊢ Option(Status) from match r2 {
237          Nil => None
238          Cons(c, r3) => Status::of_bytes(a, b, c)
239        }
240      }
241    }
242  }
243
244  // Every status reads back as itself.
245  pub theorem read_bytes(s: Status): Status::read(Status::bytes(s)) = Some(s) {
246    match s {
247      Mk(l, t, o) => Eq::trans(
248        Eq::congr(
249          ⟦Status::of_parts(□, Digit::of_byte(Digit::byte(t)), Digit::of_byte(Digit::byte(o)))⟧,
250          Lead::of_byte_byte(l)
251        ),
252        Eq::trans(
253          Eq::congr(
254            ⟦Status::of_parts(Some(l), □, Digit::of_byte(Digit::byte(o)))⟧,
255            Digit::of_byte_byte(t)
256          ),
257          Eq::congr(⟦Status::of_parts(Some(l), Some(t), □)⟧, Digit::of_byte_byte(o))
258        )
259      )
260    }
261  }
