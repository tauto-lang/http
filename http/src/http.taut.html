13  Delete: Method
14  Patch: Method

16  Options: Method

19pub module Method {
20  pub fn text(m: Method): Text {
21    match m {
22      Get => "GET"
23      Post => "POST"
24      Put => "PUT"
25      Delete => "DELETE"
26      Patch => "PATCH"
27      Head => "HEAD"
28      Options => "OPTIONS"
29    }
30  }
31
32  pub fn bytes(m: Method): List(Nat) {
33    Text::bytes(Method::text(m))
34  }
35
36  pub fn try(m: Method, b: List(Nat), other: Option(Method)): Option(Method) {
37    match byte_list_eq(Method::bytes(m), b) {
38      true => Some(m)
39      false => other
40    }
41  }
42
43  // The method a request line names. Method names are case-sensitive.
44  pub fn of(b: List(Nat)): Option(Method) {
45    Method::try(
46      Get,
47      b,
48      Method::try(
49        Post,
50        b,
51        Method::try(
52          Put,
53          b,
54          Method::try(
55            Delete,
56            b,
57            Method::try(Patch, b, Method::try(Head, b, Method::try(Options, b, None)))
58          )
59        )
60      )
61    )
62  }
63
64  pub theorem of_bytes(m: Method): Method::of(Method::bytes(m)) = Some(m) {
65    match m {
66      Get => Refl
67      Post => Refl
68      Put => Refl
69      Delete => Refl
70      Patch => Refl
71      Head => Refl
72      Options => Refl
73    }
74  }
75
76  pub fn index(m: Method): Nat {
77    match m { Get => 0, Post => 1, Put => 2, Delete => 3, Patch => 4, Head => 5, Options => 6 }
78  }
79
80  pub fn eq(a: Method, b: Method): Bool {
81    Nat::eq(Method::index(a), Method::index(b))
82  }
83
84  pub theorem eq_refl(m: Method): Method::eq(m, m) = true {
85    Nat::eq_refl(Method::index(m))
86  }
87}

89pub module Header {
90  // A header is a name and a value. Names compare without ASCII case.
91  pub fn name(h: @name Text × @value Text): Text {
92    (h: @name Text)
93  }
94
95  pub fn value(h: @name Text × @value Text): Text {
96    (h: @value Text)
97  }
98
99  pub fn pick(y: Bool, h: @name Text × @value Text, rest: Option(Text)): Option(Text) {
100    match y {
101      true => Some((h: @value Text))
102      false => rest
103    }
104  }
105
106  // The value of the first header called `name`.
107  pub fn get(hs: List(@name Text × @value Text), name: Text): Option(Text) {
108    match hs {
109      Nil => None
110      Cons(h, t) => Header::pick(
111        bytes_eq_ci(Text::bytes((h: @name Text)), Text::bytes(name)),
112        h,
113        Header::get(t, name)
114      )
115    }
116  }
117
118  pub fn pick_all(y: Bool, h: @name Text × @value Text, rest: List(Text)): List(Text) {
119    match y {
120      true => Cons((h: @value Text), rest)
121      false => rest
122    }
123  }
124
125  // The values of all the headers called `name`, in order.
126  pub fn get_all(hs: List(@name Text × @value Text), name: Text): List(Text) {
127    match hs {
128      Nil => Nil
129      Cons(h, t) => Header::pick_all(
130        bytes_eq_ci(Text::bytes((h: @name Text)), Text::bytes(name)),
131        h,
132        Header::get_all(t, name)
133      )
134    }
135  }
136
137  pub fn first_value(vs: List(Text)): Option(Text) {
138    match vs {
139      Nil => None
140      Cons(v, rest) => Some(v)
141    }
142  }
143
144  pub theorem pick_first(
145    y: Bool,
146    h: @name Text × @value Text,
147    rest: Option(Text),
148    later: List(Text),
149    e: rest = Header::first_value(later)
150  ): Header::pick(y, h, rest) = Header::first_value(Header::pick_all(y, h, later)) {
151    match y { true => Refl, false => e }
152  }
153
154  // `get` is the first of `get_all`.
155  pub theorem get_first(
156    hs: List(@name Text × @value Text),
157    name: Text
158  ): Header::get(hs, name) = Header::first_value(Header::get_all(hs, name)) {
159    match hs {
160      Nil => Refl
161      Cons(h, t) => Header::pick_first(
162        bytes_eq_ci(Text::bytes((h: @name Text)), Text::bytes(name)),
163        h,
164        Header::get(t, name),
165        Header::get_all(t, name),
166        Header::get_first(t, name)
167      )
168    }
169  }
170
171  // A header name is one or more `tchar`s.
172  pub fn name_valid(n: Text): Bool {
173    not(Nat::eq(Text::len(n), 0)) && Text::all_bytes(n, tchar_byte)
174  }
175
176  pub fn value_valid(v: Text): Bool {
177    Text::all_bytes(v, field_byte)
178  }
179
180  pub fn valid(h: @name Text × @value Text): Bool {
181    Header::name_valid((h: @name Text)) && Header::value_valid((h: @value Text))
182  }
183
184  pub fn valid_text(h: @name Text × @value Text): Bool {
185    (
186      not(
187        Nat::eq(Text::len((h: @name Text)), 0)
188      ) && all_text_bytes((c: Nat) => tchar_byte(c), (h: @name Text))
189    ) && all_text_bytes((c: Nat) => field_byte(c), (h: @value Text))
190  }
191
192  pub theorem valid_text_eq(h: @name Text × @value Text): Header::valid_text(h) = Header::valid(h) {
193    Eq::trans(
194      Eq::congr(
195        ⟦(
196          not(Nat::eq(Text::len((h: @name Text)), 0)) && □
197        ) && all_text_bytes((c: Nat) => field_byte(c), (h: @value Text))⟧,
198        Eq::trans(
199          all_text_from_eq(
200            (c: Nat) => tchar_byte(c),
201            (h: @name Text),
202            0,
203            Text::len((h: @name Text))
204          ),
205          all_from_all((c: Nat) => tchar_byte(c), Text::bytes((h: @name Text)))
206        )
207      ),
208      Eq::congr(
209        ⟦(
210          not(
211            Nat::eq(Text::len((h: @name Text)), 0)
212          ) && List::all((c: Nat) => tchar_byte(c), Text::bytes((h: @name Text)))
213        ) && □⟧,
214        Eq::trans(
215          all_text_from_eq(
216            (c: Nat) => field_byte(c),
217            (h: @value Text),
218            0,
219            Text::len((h: @value Text))
220          ),
221          all_from_all((c: Nat) => field_byte(c), Text::bytes((h: @value Text)))
222        )
223      )
224    )
225  }
226
227  pub fn gate(y: Bool, r: HttpResponse): HttpResponse {
228    match y {
229      true => r
230      false => invalid_header_response()
231    }
232  }

330// A `tchar` of RFC 9110: a visible ASCII byte that is not one of the separators
331// ( )< > @ , ; : \ " / [ ] ? = { }. The separators are compared one by one,
332// as `List::all` over their list unfolds, so a program needs no list or closure.
333pub fn tchar_byte(c: Nat): Bool {
334  Nat::lt(
335    32,
336    c
337  ) && Bool::and(
338    Nat::lt(c, 127),
339    not(
340      Nat::eq(c, 40)
341    ) && not(
342      Nat::eq(c, 41)
343    ) && not(
344      Nat::eq(c, 60)
345    ) && not(
346      Nat::eq(c, 62)
347    ) && not(
348      Nat::eq(c, 64)
349    ) && not(
350      Nat::eq(c, 44)
351    ) && not(
352      Nat::eq(c, 59)
353    ) && not(
354      Nat::eq(c, 58)
355    ) && not(
356      Nat::eq(c, 92)
357    ) && not(
358      Nat::eq(c, 34)
359    ) && not(
360      Nat::eq(c, 47)
361    ) && not(
362      Nat::eq(c, 91)
363    ) && not(
364      Nat::eq(c, 93)
365    ) && not(
366      Nat::eq(c, 63)
367    ) && not(Nat::eq(c, 61)) && not(Nat::eq(c, 123)) && not(Nat::eq(c, 125)) && true
368  )
369}

371// A byte of a field value: tab, visible ASCII, space, or a byte of UTF-8 text.
372// Not CR, LF, NUL or another control byte, so a value cannot end its line.
373pub fn field_byte(c: Nat): Bool {
374  Nat::eq(c, 9) || (Nat::lt(31, c) && not(Nat::eq(c, 127)))
375}

381// `List::all(p, bs)` read by position: whether `p` holds for the `k` elements
382// of `bs` from `i` on.
383pub fn all_from<A: Shared>(p: (A) -> Bool, bs: List(A), i: Nat, k: Nat): Bool {
384  match k {
385    Zero => true
386    Succ(k2) => ⊢ Bool from match List::nth(bs, i) {
387      None => true
388      Some(b) => p(b) && all_from(p, bs, Succ(i), k2)
389    }
390  }
391}
392
393pub theorem all_from_shift<A: Type>(
394  p: (A) -> Bool,
395  h: A,
396  t: List(A),
397  i: Nat,
398  k: Nat
399): all_from(p, Cons(h, t), Succ(i), k) = all_from(p, t, i, k) {
400  match k {
401    Zero => Refl
402    Succ(k2) => Eq::congr(
403      (z: Bool) => (⊢ Bool from match List::nth(t, i) {
404        None => true
405        Some(b) => p(b) && z
406      }),
407      all_from_shift(p, h, t, Succ(i), k2)
408    )
409  }
410}
411
412pub theorem all_from_all<A: Type>(
413  p: (A) -> Bool,
414  bs: List(A)
415): all_from(p, bs, 0, List::length(bs)) = List::all(p, bs) {
416  match bs {
417    Nil => Refl
418    Cons(h, t) => Eq::congr(
419      ⟦p(h) && □⟧,
420      Eq::trans(all_from_shift(p, h, t, 0, List::length(t)), all_from_all(p, t))
421    )
422  }
423}

425// Whether `p` holds for every byte of `t`, read with `Text::byte`: extraction
426// reads the native text and builds no list of bytes.
427pub fn all_text_from(p: (Nat) -> Bool, t: Text, i: Nat, k: Nat): Bool {
428  match k {
429    Zero => true
430    Succ(k2) => ⊢ Bool from match Text::byte(t, i) {
431      None => true
432      Some(b) => p(b) && all_text_from(p, t, Succ(i), k2)
433    }
434  }
435}
436
437pub theorem all_text_from_eq(
438  p: (Nat) -> Bool,
439  t: Text,
440  i: Nat,
441  k: Nat
442): all_text_from(p, t, i, k) = all_from(p, Text::bytes(t), i, k) {
443  match k {
444    Zero => Refl
445    Succ(k2) => Eq::congr(
446      (z: Bool) => (⊢ Bool from match List::nth(Text::bytes(t), i) {
447        None => true
448        Some(b) => p(b) && z
449      }),
450      all_text_from_eq(p, t, Succ(i), k2)
451    )
452  }
453}

455pub fn all_text_bytes(p: (Nat) -> Bool, t: Text): Bool {

467  hs: List(@name Text × @value Text)
468): headers_valid_text(hs) = headers_valid(hs) {

478// What goes out in place of a response with a header that may not be sent: a
479// bad name or value is a bug in the server, and sending it could split the response.
480pub fn invalid_header_response(): HttpResponse {
481  HttpResponse::Mk(
482    Status::internal_server_error(),
483    Cons(#(@name "content-type", @value "text/plain; charset=utf-8"), Nil),
484    "invalid response header"
485  )
486}

488// The same response with one more header, after the others. A name that is
489// not a token, or a value with CR, LF or another control byte, turns the
490// response into `invalid_header_response()`.
491pub fn with_header(r: HttpResponse, name: Text, value: Text): HttpResponse {
492  match r {
493    Mk(s, hs, b) => Header::gate(
494      Header::name_valid(name) && Header::value_valid(value),
495      HttpResponse::Mk(s, List::append(hs, Cons(#(@name name, @value value), Nil)), b)
496    )
497  }
498}

735pub fn reason_pick(y: Bool, here: Text, later: Text): Text {
736  match y { true => here, false => later }
737}

739pub fn reason_in(s: Nat, l: List(Nat × Text)): Text {
740  match l {
741    Nil => ""
742    Cons(p, t) => reason_pick(Nat::eq((p: Nat), s), (p: Text), reason_in(s, t))
743  }
744}
