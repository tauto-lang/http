38// HTTP/1.1 on the wire: reading a request and writing a response.
39//
40//   request  = method SP target SP "HTTP/1.1" CRLF *(name ":" SP value CRLF) CRLF body
41//   target   = path [ "?" query ]
42//
43// The body is everything after the empty line: one request per connection, as
44// with `Connection: close`. The reader gives back what the printer wrote
45// (`parse_print_request`) for every request whose parts hold no byte that
46// would end them early (`HttpRequest::ok`).
47
48// ---- printing a request --------------------------------------------------------------
49
50pub module Query {
51  pub fn bytes(q: Option(Text)): List(Nat) {
52    match q {
53      None => Nil
54      Some(t) => Cons(63, Text::bytes(t))
55    }
56  }
57
58  pub fn of(r: List(Nat)): Option(Text) {
59    match r {
60      Nil => None
61      Cons(c, t) => Some(Text::Mk(t))
62    }
63  }
64}
65
66pub fn target_bytes(p: Text, q: Option(Text)): List(Nat) {
67  List::append(Text::bytes(p), Query::bytes(q))
68}
69
70// `HTTP/1.1` and the carriage return that ends the request line.
71pub fn version_tail(): List(Nat) {
72  Text::bytes("HTTP/1.1\r")
73}

75pub module Start {
76  // The request line, without its line feed.
77  pub fn line(m: Method, p: Text, q: Option(Text)): List(Nat) {
78    List::append(
79      Method::bytes(m),
80      Cons(32, List::append(target_bytes(p, q), Cons(32, version_tail())))
81    )
82  }
83
84  pub fn done(m: Method, t: List(Nat), y: Bool): Option(Start) {
85    match y {
86      true => start_of(m, take_until(63, t))
87      false => None
88    }
89  }
90
91  pub fn rest(m: Method, p: @before List(Nat) × @after List(Nat)): Option(Start) {
92    match p {
93      #(@before b, @after a) => Start::done(m, b, byte_list_eq(a, Cons(32, version_tail())))
94    }
95  }
96
97  // `r` is empty or starts with the space that ended the method.
98  pub fn target(o: Option(Method), r: List(Nat)): Option(Start) {
99    match o {
100      None => None
101      Some(m) => ⊢ Option(Start) from match r {
102        Nil => None
103        Cons(c, x) => Start::rest(m, take_until(32, x))
104      }
105    }
106  }
107
108  pub fn method(p: @before List(Nat) × @after List(Nat)): Option(Start) {
109    match p {
110      #(@before b, @after a) => Start::target(Method::of(b), a)
111    }
112  }
113
114  pub theorem of_target(
115    m: Method,
116    pb: List(Nat),
117    q: Option(Text),
118    ok: no_byte(63, pb) = true
119  ): start_of(
120    m,
121    take_until(63, List::append(pb, Query::bytes(q)))
122  ) = Some(Start::Mk(m, Text::Mk(pb), q)) {
123    ⊢ start_of(
124      m,
125      take_until(63, List::append(pb, Query::bytes(q)))
126    ) = Some(Start::Mk(m, Text::Mk(pb), q)) from match q {
127      None => Eq::congr(
128        ⟦start_of(m, □)⟧,
129        Eq::trans(Eq::congr(⟦take_until(63, □)⟧, List::append_nil(pb)), take_until_none(63, pb, ok))
130      )
131      Some(qt) => ⊢ start_of(
132        m,
133        take_until(63, List::append(pb, Query::bytes(Some(qt))))
134      ) = Some(Start::Mk(m, Text::Mk(pb), Some(qt))) from match qt {
135        Text::Mk(qb) => Eq::congr(⟦start_of(m, □)⟧, take_until_append(63, pb, qb, ok))
136      }
137    }
138  }
139
140  pub theorem line_no_lf(
141    m: Method,
142    p: Text,
143    q: Option(Text),
144    ok: no_byte(10, target_bytes(p, q)) = true
145  ): no_byte(10, Start::line(m, p, q)) = true {
146    Eq::trans(
147      no_byte_append(
148        10,
149        Method::bytes(m),
150        Cons(32, List::append(target_bytes(p, q), Cons(32, version_tail())))
151      ),
152      Bool::and_intro(
153        no_byte(10, Method::bytes(m)),
154        no_byte(10, Cons(32, List::append(target_bytes(p, q), Cons(32, version_tail())))),
155        Method::no_lf(m),
156        Eq::trans(
157          no_byte_append(10, target_bytes(p, q), Cons(32, version_tail())),
158          Bool::and_intro(
159            no_byte(10, target_bytes(p, q)),
160            no_byte(10, Cons(32, version_tail())),
161            ok,
162            Refl
163          )
164        )
165      )
166    )
167  }

170pub module Header {
171  // `name: value` and a carriage return.
172  pub fn line(h: @name Text × @value Text): List(Nat) {
173    List::append(
174      Text::bytes((h: @name Text)),
175      Cons(58, Cons(32, List::append(Text::bytes((h: @value Text)), Cons(13, Nil))))
176    )
177  }
178
179  // The name has no `:` or line feed; the value has no line feed.
180  pub fn ok(h: @name Text × @value Text): Bool {
181    Bool::and(
182      no_byte(58, Text::bytes((h: @name Text))),
183      no_byte(10, Text::bytes((h: @name Text))) && no_byte(10, Text::bytes((h: @value Text)))
184    )
185  }
186
187  pub fn cr(n: List(Nat), o: Option(List(Nat))): Option(@name Text × @value Text) {
188    match o {
189      None => None
190      Some(v) => Some(#(@name Text::Mk(n), @value Text::Mk(v)))
191    }
192  }
193
194  pub fn value_of(n: List(Nat), o: Option(List(Nat))): Option(@name Text × @value Text) {
195    match o {
196      None => None
197      Some(v) => Header::cr(n, strip_cr(v))
198    }
199  }
200
201  pub fn name_of(p: @before List(Nat) × @after List(Nat)): Option(@name Text × @value Text) {
202    match p {
203      #(@before b, @after a) => Header::value_of(b, strip_prefix(Cons(58, Cons(32, Nil)), a))
204    }
205  }
206
207  pub theorem line_no_lf(
208    h: @name Text × @value Text,
209    ok: Header::ok(h) = true
210  ): no_byte(10, Header::line(h)) = true {
211    Eq::trans(
212      no_byte_append(
213        10,
214        Text::bytes((h: @name Text)),
215        Cons(58, Cons(32, List::append(Text::bytes((h: @value Text)), Cons(13, Nil))))
216      ),
217      Bool::and_intro(
218        no_byte(10, Text::bytes((h: @name Text))),
219        no_byte(10, Cons(58, Cons(32, List::append(Text::bytes((h: @value Text)), Cons(13, Nil))))),
220        yes_left(
221          no_byte(10, Text::bytes((h: @name Text))),
222          no_byte(10, Text::bytes((h: @value Text))),
223          yes_right(
224            no_byte(58, Text::bytes((h: @name Text))),
225            no_byte(10, Text::bytes((h: @name Text))) && no_byte(10, Text::bytes((h: @value Text))),
226            ok
227          )
228        ),
229        Eq::trans(
230          no_byte_append(10, Text::bytes((h: @value Text)), Cons(13, Nil)),
231          Bool::and_intro(
232            no_byte(10, Text::bytes((h: @value Text))),
233            no_byte(10, Cons(13, Nil)),
234            yes_right(
235              no_byte(10, Text::bytes((h: @name Text))),
236              no_byte(10, Text::bytes((h: @value Text))),
237              yes_right(
238                no_byte(58, Text::bytes((h: @name Text))),
239                no_byte(
240                  10,
241                  Text::bytes((h: @name Text))
242                ) && no_byte(10, Text::bytes((h: @value Text))),
243                ok
244              )
245            ),
246            Refl
247          )
248        )
249      )
250    )
251  }

253  pub theorem line_not_blank(h: @name Text × @value Text): is_blank(Header::line(h)) = false {
254    match h {
255      #(@name n, @value v) => ⊢ is_blank(Header::line(#(@name n, @value v))) = false from match n {
256        Text::Mk(nb) => ⊢ is_blank(
257          Header::line(#(@name Text::Mk(nb), @value v))
258        ) = false from match nb {
259          Nil => Refl
260          Cons(x, t) => ⊢ is_blank(
261            Header::line(#(@name Text::Mk(Cons(x, t)), @value v))
262          ) = false from match t {
263            Nil => Refl
264            Cons(y, u) => Refl
265          }
266        }
267      }
268    }
269  }

364pub fn headers_step(
365  y: Bool,
366  h: Option(@name Text × @value Text),
367  body: List(Nat),
368  rest: Option(List(@name Text × @value Text) × List(Nat))
369): Option(List(@name Text × @value Text) × List(Nat)) {
370  match y {
371    true => Some(#(Nil, body))
372    false => ⊢ Option(List(@name Text × @value Text) × List(Nat)) from match h {
373      None => None
374      Some(x) => cons_headers(x, rest)
375    }
376  }

379// The headers up to the empty line, and the body after it.
380pub fn parse_headers(ls: List(List(Nat))): Option(List(@name Text × @value Text) × List(Nat)) {
381  match ls {
382    Nil => None
383    // The same as `headers_step(is_blank(l), parse_header(l), unlines(more), parse_headers(more))`,
384    // written out so that a program computes the body and the later headers only when it needs them.
385    Cons(l, more) => ⊢ Option(List(@name Text × @value Text) × List(Nat)) from match is_blank(l) {
386      true => Some(#(Nil, unlines(more)))
387      false => ⊢ Option(List(@name Text × @value Text) × List(Nat)) from match parse_header(l) {
388        None => None
389        Some(x) => cons_headers(x, parse_headers(more))
390      }
391    }
392  }
393}

407      }

452pub theorem parse_header_line(
453  h: @name Text × @value Text,
454  ok: Header::ok(h) = true
455): parse_header(Header::line(h)) = Some(h) {
456  match h {
457    #(@name n, @value v) => ⊢ parse_header(
458      Header::line(#(@name n, @value v))
459    ) = Some(#(@name n, @value v)) from match n {
460      Text::Mk(nb) => ⊢ parse_header(
461        Header::line(#(@name Text::Mk(nb), @value v))
462      ) = Some(#(@name Text::Mk(nb), @value v)) from match v {
463        Text::Mk(vb) => Eq::trans(
464          Eq::congr(
465            ⟦Header::name_of(□)⟧,
466            take_until_append(
467              58,
468              nb,
469              Cons(32, List::append(vb, Cons(13, Nil))),
470              yes_left(no_byte(58, nb), no_byte(10, nb) && no_byte(10, vb), ok)
471            )
472          ),
473          Eq::congr(⟦Header::cr(nb, □)⟧, strip_cr_append(vb))
474        )
475      }
476    }
477  }
478}

480pub theorem parse_head_lines(
481  hs: List(@name Text × @value Text),
482  body: List(Nat),
483  ok: headers_ok(hs) = true
484): parse_headers(lines(Head::lines(hs, body))) = Some(#(hs, body)) {
485  match hs {
486    Nil => Eq::trans(
487      Eq::congr(⟦parse_headers(□)⟧, lines_append(Cons(13, Nil), body, Refl)),
488      Eq::congr(
489        ⟦the(Option(List(@name Text × @value Text) × List(Nat)), Some(#(Nil, □)))⟧,
490        unlines_lines(body)
491      )
492    )
493    Cons(h, t) => Eq::trans(
494      Eq::congr(
495        ⟦parse_headers(□)⟧,
496        lines_append(
497          Header::line(h),
498          Head::lines(t, body),
499          Header::line_no_lf(h, yes_left(Header::ok(h), headers_ok(t), ok))
500        )
501      ),
502      Eq::trans(
503        Eq::congr(
504          ⟦headers_step(
505            □,
506            parse_header(Header::line(h)),
507            unlines(lines(Head::lines(t, body))),
508            parse_headers(lines(Head::lines(t, body)))
509          )⟧,
510          Header::line_not_blank(h)
511        ),
512        Eq::trans(
513          Eq::congr(
514            ⟦headers_step(
515              false,
516              □,
517              unlines(lines(Head::lines(t, body))),
518              parse_headers(lines(Head::lines(t, body)))
519            )⟧,
520            parse_header_line(h, yes_left(Header::ok(h), headers_ok(t), ok))
521          ),
522          Eq::congr(
523            ⟦headers_step(false, Some(h), unlines(lines(Head::lines(t, body))), □)⟧,
524            parse_head_lines(t, body, yes_right(Header::ok(h), headers_ok(t), ok))
525          )
526        )
527      )
528    )
529  }

532pub theorem parse_start_line(
533  m: Method,
534  pb: List(Nat),
535  q: Option(Text),
536  ok: target_ok(Text::Mk(pb), q) = true
537): parse_start(Start::line(m, Text::Mk(pb), q)) = Some(Start::Mk(m, Text::Mk(pb), q)) {
538  Eq::trans(
539    Eq::congr(
540      ⟦Start::method(□)⟧,
541      take_until_append(
542        32,
543        Method::bytes(m),
544        List::append(target_bytes(Text::Mk(pb), q), Cons(32, version_tail())),
545        Method::no_sp(m)
546      )
547    ),
548    Eq::trans(
549      Eq::congr(
550        ⟦Start::target(
551          □,
552          Cons(32, List::append(target_bytes(Text::Mk(pb), q), Cons(32, version_tail())))
553        )⟧,
554        Method::of_bytes(m)
555      ),
556      Eq::trans(
557        Eq::congr(
558          ⟦Start::rest(m, □)⟧,
559          take_until_append(
560            32,
561            target_bytes(Text::Mk(pb), q),
562            version_tail(),
563            yes_left(
564              no_byte(32, target_bytes(Text::Mk(pb), q)),
565              no_byte(10, target_bytes(Text::Mk(pb), q)),
566              yes_right(
567                no_byte(63, pb),
568                no_byte(
569                  32,
570                  target_bytes(Text::Mk(pb), q)
571                ) && no_byte(10, target_bytes(Text::Mk(pb), q)),
572                ok
573              )
574            )
575          )
576        ),
577        Start::of_target(
578          m,
579          pb,
580          q,
581          yes_left(
582            no_byte(63, pb),
583            no_byte(32, target_bytes(Text::Mk(pb), q)) && no_byte(
584              10,
585              target_bytes(Text::Mk(pb), q)
586            ),
587            ok
588          )
589        )
590      )
591    )
592  )

595// Reading a printed request gives it back.
596pub theorem parse_print_request(
597  r: HttpRequest,
598  ok: HttpRequest::ok(r) = true
599): parse_request(print_request(r)) = Some(r) {
600  match r {
601    Mk(m, p, q, hs, b) => ⊢ parse_request(
602      print_request(HttpRequest::Mk(m, p, q, hs, b))
603    ) = Some(HttpRequest::Mk(m, p, q, hs, b)) from match p {
604      Text::Mk(pb) => ⊢ parse_request(
605        print_request(HttpRequest::Mk(m, Text::Mk(pb), q, hs, b))
606      ) = Some(HttpRequest::Mk(m, Text::Mk(pb), q, hs, b)) from match b {
607        Text::Mk(bb) => Eq::trans(
608          Eq::congr(
609            ⟦parse_lines(□)⟧,
610            lines_append(
611              Start::line(m, Text::Mk(pb), q),
612              Head::lines(hs, bb),
613              Start::line_no_lf(
614                m,
615                Text::Mk(pb),
616                q,
617                yes_right(
618                  no_byte(32, target_bytes(Text::Mk(pb), q)),
619                  no_byte(10, target_bytes(Text::Mk(pb), q)),
620                  yes_right(
621                    no_byte(63, pb),
622                    no_byte(
623                      32,
624                      target_bytes(Text::Mk(pb), q)
625                    ) && no_byte(10, target_bytes(Text::Mk(pb), q)),
626                    yes_left(target_ok(Text::Mk(pb), q), headers_ok(hs), ok)
627                  )
628                )
629              )
630            )
631          ),
632          Eq::trans(
633            Eq::congr(
634              ⟦build_request(□, parse_headers(lines(Head::lines(hs, bb))))⟧,
635              parse_start_line(m, pb, q, yes_left(target_ok(Text::Mk(pb), q), headers_ok(hs), ok))
636            ),
637            Eq::congr(
638              ⟦build_request(Some(Start::Mk(m, Text::Mk(pb), q)), □)⟧,
639              parse_head_lines(hs, bb, yes_right(target_ok(Text::Mk(pb), q), headers_ok(hs), ok))
640            )
641          )
642        )
643      }
644    }
645  }

648// ---- writing a response --------------------------------------------------------------
649//
650//   response = "HTTP/1.1" SP status SP reason CRLF *(name ":" SP value CRLF) CRLF body
651//
652// The status is three digits. After the response's own headers, less any
653// `content-length` among them, comes the `content-length` of its body.
654
655pub fn digit_byte(d: Nat): Nat {
656  Nat::add(d, 48)
657}
658
659// The three digits of a status.
660pub fn status_digits(s: Status): List(Nat) {
661  Status::bytes(s)
662}
663
664pub fn version_head(): List(Nat) {
665  Text::bytes("HTTP/1.1 ")
666}
667
668// The status line, without its line feed.
669pub fn status_line_bytes(s: Status): List(Nat) {
670  List::append(
671    version_head(),
672    List::append(status_digits(s), Cons(32, List::append(Text::bytes(reason(s)), Cons(13, Nil))))
673  )

676pub fn content_length(b: Text): @name Text × @value Text {
677  #(@name "content-length", @value Text::of_nat(Text::len(b)))
678}

680// Whether a header is called `content-length`, whatever the case of its name.
681pub fn is_content_length(h: @name Text × @value Text): Bool {
682  bytes_eq_ci(Text::bytes((h: @name Text)), Text::bytes("content-length"))
683}
684
685pub fn keep_unless(
686  y: Bool,
687  h: @name Text × @value Text,
688  rest: List(@name Text × @value Text)
689): List(@name Text × @value Text) {
690  match y {
691    true => rest
692    false => Cons(h, rest)
693  }
694}

696// The headers without any `content-length`.
697pub fn without_content_length(hs: List(@name Text × @value Text)): List(@name Text × @value Text) {
698  match hs {
699    Nil => Nil
700    Cons(h, t) => keep_unless(is_content_length(h), h, without_content_length(t))
701  }
702}

704// Whether a header is called `transfer-encoding`, whatever the case of its name.
705pub fn is_transfer_encoding(h: @name Text × @value Text): Bool {
706  bytes_eq_ci(Text::bytes((h: @name Text)), Text::bytes("transfer-encoding"))
707}
708
709// The headers without any `transfer-encoding`.
710pub fn without_transfer_encoding(
711  hs: List(@name Text × @value Text)
712): List(@name Text × @value Text) {
713  match hs {
714    Nil => Nil
715    Cons(h, t) => keep_unless(is_transfer_encoding(h), h, without_transfer_encoding(t))
716  }
717}

719// The headers without the ones that say how the body is framed: neither a
720// `content-length` nor a `transfer-encoding`.
721pub fn without_framing(hs: List(@name Text × @value Text)): List(@name Text × @value Text) {
722  without_content_length(without_transfer_encoding(hs))
723}

725// The headers as they go out: the response's own without any `content-length` or
726// `transfer-encoding` of its own (the names are compared without ASCII case), then the
727// one `content-length` of its body. The body goes out by its length, never chunked.
728pub fn sent_headers(hs: List(@name Text × @value Text), b: Text): List(@name Text × @value Text) {
729  List::append(without_framing(hs), Cons(content_length(b), Nil))

732pub module HttpResponse {
733  pub fn bytes(r: HttpResponse): List(Nat) {
734    match r {
735      Mk(s, hs, b) => List::append(
736        status_line_bytes(s),
737        Cons(10, Head::lines(sent_headers(hs, b), Text::bytes(b)))
738      )
739    }
740  }
741
742  pub fn ok(r: HttpResponse): Bool {
743    match r {
744      Mk(s, hs, b) => status_ok(s) && headers_ok(hs)
745    }
746  }
747}

749// The response itself if every header may be sent, otherwise
750// `invalid_header_response()`. A response built with `response(..)` directly
751// is checked here too, so no header reaches the wire with a line break in it.
752pub fn checked_response(r: HttpResponse): HttpResponse {
753  Header::gate(headers_valid(HttpResponse::headers(r)), r)

756pub theorem checked_response_valid(
757  r: HttpResponse,
758  v: headers_valid(HttpResponse::headers(r)) = true

817// Three digits and a space; the reason after them is not read.
818pub fn status_of(o: Option(List(Nat))): Option(Status) {
819  match o {
820    None => None
821    Some(l) => ⊢ Option(Status) from match l {
822      Nil => None
823      Cons(a, l1) => ⊢ Option(Status) from match l1 {
824        Nil => None
825        Cons(b, l2) => ⊢ Option(Status) from match l2 {
826          Nil => None
827          Cons(c, l3) => ⊢ Option(Status) from match l3 {
828            Nil => None
829            Cons(sp, rest) => status_space(Nat::eq(sp, 32), a, b, c)
830          }
831        }
832      }
833    }
834  }
835}

878): parse_status(status_line_bytes(s)) = Some(s) {

880    Eq::congr(

905pub theorem reason_in_no_lf(
906  s: Nat,
907  l: List(Nat × Text),
908  h: reasons_ok(l) = true
909): no_byte(10, Text::bytes(reason_in(s, l))) = true {
910  match l {
911    Nil => Refl
912    Cons(#(k, v), t) => ⊢ no_byte(
913      10,
914      Text::bytes(reason_pick(Nat::eq(k, s), v, reason_in(s, t)))
915    ) = true from match Nat::eq(k, s) eqn e {
916      false => reason_in_no_lf(s, t, yes_right(no_byte(10, Text::bytes(v)), reasons_ok(t), h))
917      true => yes_left(no_byte(10, Text::bytes(v)), reasons_ok(t), h)
918    }
919  }
920}

922pub theorem status_line_no_lf(s: Status): no_byte(10, status_line_bytes(s)) = true {
923  Eq::trans(
924    no_byte_append(
925      10,
926      version_head(),
927      List::append(status_digits(s), Cons(32, List::append(Text::bytes(reason(s)), Cons(13, Nil))))
928    ),
929    Bool::and_intro(
930      no_byte(10, version_head()),
931      no_byte(
932        10,
933        List::append(
934          status_digits(s),
935          Cons(32, List::append(Text::bytes(reason(s)), Cons(13, Nil)))
936        )
937      ),
938      Refl,
939      Eq::trans(
940        no_byte_append(
941          10,
942          status_digits(s),
943          Cons(32, List::append(Text::bytes(reason(s)), Cons(13, Nil)))
944        ),
945        Bool::and_intro(
946          no_byte(10, status_digits(s)),
947          no_byte(10, Cons(32, List::append(Text::bytes(reason(s)), Cons(13, Nil)))),
948          status_digits_no_lf(s),
949          Eq::trans(
950            no_byte_append(10, Text::bytes(reason(s)), Cons(13, Nil)),
951            Bool::and_intro(
952              no_byte(10, Text::bytes(reason(s))),
953              no_byte(10, Cons(13, Nil)),
954              reason_in_no_lf(Status::code(s), reasons(), Refl),
955              Refl
956            )
957          )
958        )
959      )
960    )
961  )
962}

964// No digit of a status is a line feed.
965pub theorem status_digits_no_lf(s: Status): no_byte(10, status_digits(s)) = true {
966  match s {
967    Mk(l, t, o) => Eq::trans(
968      Eq::congr(
969        ⟦not(□) && (
970          not(Nat::eq(Digit::byte(t), 10)) && (not(Nat::eq(Digit::byte(o), 10)) && true)
971        )⟧,
972        Lead::byte_not_lf(l)
973      ),
974      Eq::trans(
975        Eq::congr(
976          ⟦not(false) && (not(□) && (not(Nat::eq(Digit::byte(o), 10)) && true))⟧,
977          Digit::byte_not_lf(t)
978        ),
979        Eq::congr(⟦not(false) && (not(false) && (not(□) && true))⟧, Digit::byte_not_lf(o))
980      )
981    )
982  }
983}

986pub fn digits_pick(y: Bool, f: Nat, n: Nat, acc: List(Nat)): List(Nat) {
987  match y {
988    true => Cons(Nat::add(Nat::rem_nz(n, 10, Nat::succ_ne_zero(9)), 48), acc)
989    false => Nat::digits(
990      f,
991      Nat::div_nz(n, 10, Nat::succ_ne_zero(9)),
992      Cons(Nat::add(Nat::rem_nz(n, 10, Nat::succ_ne_zero(9)), 48), acc)
993    )
994  }

997pub theorem nat_digits_no_lf(
998  f: Nat,
999  n: Nat,
1000  acc: List(Nat),
1001  h: no_byte(10, acc) = true
1002): no_byte(10, Nat::digits(f, n, acc)) = true {
1003  match f {
1004    Zero => h
1005    Succ(g) => ⊢ no_byte(
1006      10,
1007      digits_pick(Nat::eq(Nat::div_nz(n, 10, Nat::succ_ne_zero(9)), 0), g, n, acc)
1008    ) = true from match Nat::eq(Nat::div_nz(n, 10, Nat::succ_ne_zero(9)), 0) eqn e {
1009      false => nat_digits_no_lf(
1010        g,
1011        Nat::div_nz(n, 10, Nat::succ_ne_zero(9)),
1012        Cons(Nat::add(Nat::rem_nz(n, 10, Nat::succ_ne_zero(9)), 48), acc),
1013        h
1014      )
1015      true => h
1016    }
1017  }

1211  ok: HttpResponse::ok(r) = true

1227              □,

1229            )⟧,

1247              )
1248            )
1249          )
1250        )
1251      )
1252    }
