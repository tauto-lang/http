63// Readers that eat a message a few bytes at a time, as they come off a
64// connection: nothing waits for the whole message, and a body passes on as it
65// arrives. A reader is one step on one byte; `feed` runs it over the bytes at
66// hand. `feed_append` says that how the bytes were cut into pieces does not
67// change what the reader decides or passes on.
68
69// ---- feeding bytes to a reader -------------------------------------------------------
70
71// What feeding leaves: the reader's state, the bytes it passed on, and the
72// bytes after the reader stopped (they belong to whatever comes next).
73pub(open) inductive Fed(S: Type): Type {
74  Mk(state: S, out: List(Nat), rest: List(Nat)): Fed(S)
75}
76
77pub fn fed_state<S>(f: Fed(S)): S {
78  match f {
79    Fed::Mk(s, o, r) => s
80  }
81}
82
83pub fn fed_out<S>(f: Fed(S)): List(Nat) {
84  match f {
85    Fed::Mk(s, o, r) => o
86  }
87}

89pub fn fed_rest<S>(f: Fed(S)): List(Nat) {
90  match f {
91    Fed::Mk(s, o, r) => r
92  }

95// `f` with `o` passed on before what it passed on.
96pub fn fed_prepend<S>(o: List(Nat), f: Fed(S)): Fed(S) {
97  match f {
98    Fed::Mk(s, o2, r) => Fed::Mk(s, List::append(o, o2), r)
99  }

128// One step: `None` when the reader has stopped and the byte is not its own;
129// otherwise the next state and the bytes it passes on.
130pub fn feed_pick<S: Shared>(
131  o: Option(@state S × @rest List(Nat)),
132  s: S,
133  xs: List(Nat),
134  k: (S) -> Fed(S)
135): Fed(S) {
136  match o {
137    None => Fed::Mk(s, Nil, xs)
138    Some(p) => fed_prepend(step_rest(p), k(step_state(p)))
139  }
140}

142// Runs `step` over `xs` from `s` until the bytes run out or the reader stops.
143pub fn feed<S: Shared>(
144  step: (S, Nat) -> Option(@state S × @rest List(Nat)),
145  s: S,
146  xs: List(Nat)
147): Fed(S) {
148  match xs {
149    Nil => Fed::Mk(s, Nil, Nil)
150    // `feed_pick(step(s, b), s, Cons(b, t), (s2: S) => feed(step, s2, t))`, written out
151    // so that a program makes no closure for each byte.
152    Cons(b, t) => ⊢ Fed(S) from match step(s, b) {
153      None => Fed::Mk(s, Nil, Cons(b, t))
154      Some(p) => fed_prepend(step_rest(p), feed(step, step_state(p), t))
155    }
156  }
157}

159// Feeding `ys` after feeding gave `f`: when the reader stopped inside the
160// first piece, `ys` joins what was left over; otherwise the reader goes on.
161pub fn feed_then<S: Shared>(
162  step: (S, Nat) -> Option(@state S × @rest List(Nat)),
163  f: Fed(S),
164  ys: List(Nat)
165): Fed(S) {
166  match f {
167    Fed::Mk(s, o, r) => ⊢ Fed(S) from match r {
168      Nil => fed_prepend(o, feed(step, s, ys))
169      Cons(b, t) => Fed::Mk(s, o, List::append(Cons(b, t), ys))
170    }
171  }
172}

180pub theorem fed_prepend_prepend<S: Type>(
181  a: List(Nat),
182  b: List(Nat),
183  f: Fed(S)
184): fed_prepend(a, fed_prepend(b, f)) = fed_prepend(List::append(a, b), f) {
185  match f {
186    Fed::Mk(s, o, r) => Eq::congr(
187      ⟦the(Fed(S), Fed::Mk(s, □, r))⟧,
188      Eq::sym(List::append_assoc(a, b, o))
189    )
190  }

193pub theorem feed_then_prepend<S: Type>(
194  step: (S, Nat) -> Option(@state S × @rest List(Nat)),
195  o: List(Nat),
196  f: Fed(S),
197  ys: List(Nat)
198): feed_then(step, fed_prepend(o, f), ys) = fed_prepend(o, feed_then(step, f, ys)) {
199  match f {
200    Fed::Mk(s, o2, r) => ⊢ feed_then(
201      step,
202      fed_prepend(o, Fed::Mk(s, o2, r)),
203      ys
204    ) = fed_prepend(o, feed_then(step, Fed::Mk(s, o2, r), ys)) from match r {
205      Nil => Eq::sym(fed_prepend_prepend(o, o2, feed(step, s, ys)))
206      Cons(b, t) => Refl
207    }
208  }

211pub theorem feed_pick_append<S: Type>(
212  step: (S, Nat) -> Option(@state S × @rest List(Nat)),
213  o: Option(@state S × @rest List(Nat)),
214  s: S,
215  b: Nat,
216  t: List(Nat),
217  ys: List(Nat),
218  ih: (s2: S) -> feed(step, s2, List::append(t, ys)) = feed_then(step, feed(step, s2, t), ys)
219): feed_pick(
220  o,
221  s,
222  Cons(b, List::append(t, ys)),
223  (s2: S) => feed(step, s2, List::append(t, ys))
224) = feed_then(step, feed_pick(o, s, Cons(b, t), (s2: S) => feed(step, s2, t)), ys) {
225  ⊢ feed_pick(
226    o,
227    s,
228    Cons(b, List::append(t, ys)),
229    (s2: S) => feed(step, s2, List::append(t, ys))
230  ) = feed_then(step, feed_pick(o, s, Cons(b, t), (s2: S) => feed(step, s2, t)), ys) from match o {
231    None => Refl
232    Some(p) => Eq::trans(
233      Eq::congr(⟦fed_prepend(step_rest(p), □)⟧, ih(step_state(p))),
234      Eq::sym(feed_then_prepend(step, step_rest(p), feed(step, step_state(p), t), ys))
235    )
236  }
237}

239// Feeding two pieces one after the other is feeding them joined.
240pub theorem feed_append<S: Type>(
241  step: (S, Nat) -> Option(@state S × @rest List(Nat)),
242  s: S,
243  xs: List(Nat),
244  ys: List(Nat)
245): feed(step, s, List::append(xs, ys)) = feed_then(step, feed(step, s, xs), ys) {
246  match xs {
247    Nil => Eq::sym(fed_prepend_nil(feed(step, s, ys)))
248    Cons(b, t) => feed_pick_append(
249      step,
250      step(s, b),
251      s,
252      b,
253      t,
254      ys,
255      (s2: S) => feed_append(step, s2, t, ys)
256    )
257  }

589      }
590    }

1723  // ---- how the body is delimited -------------------------------------------------------
1724  //
1725  // RFC 9112, section 6. A reader that guesses where an ambiguous body ends can
1726  // disagree with the next reader on the path, and the bytes one of them takes
1727  // as the next request are the other's body (request smuggling). So every
1728  // ambiguity is refused: Content-Length values that are not all the same
1729  // number, Transfer-Encoding together with Content-Length, and transfer
1730  // codings this reader does not know.
1731
1732  pub fn framing_of(max_body: Nat, c: Coding, l: Length): Framing {
1733    match c {
1734      NoCoding => ⊢ Framing from match l {
1735        NoLength => NoBody
1736        LengthIs(n) => sized_within(Nat::le(n, max_body), n)
1737        LengthBad => Refused(Status::bad_request())
1738      }
1739      JustChunked => ⊢ Framing from match l {
1740        NoLength => Chunked
1741        LengthIs(n) => Refused(Status::bad_request())
1742        LengthBad => Refused(Status::bad_request())
1743      }
1744      OtherChunked => Refused(Status::not_implemented())
1745      NotChunked => Refused(Status::bad_request())
1746    }
1747  }

1749  // A request's body, refused with 413 when it is longer than `max_body`.
1750  pub fn framing(max_body: Nat, hs: List(@name Text × @value Text)): Framing {
1751    HttpRequest::framing_of(max_body, coding_of(hs), content_length_of(hs))
1752  }

1764        LengthIs(k) => Eq::congr(LengthIs, sized_within_is(Nat::le(k, max_body), k, n, e))
1765        LengthBad => False::elim(false_ne_true(Eq::congr(is_sized, e)))
1766      }
1767      JustChunked => ⊢ l = LengthIs(n) from match l {
1768        NoLength => False::elim(false_ne_true(Eq::congr(is_sized, e)))
1769        LengthIs(k) => False::elim(false_ne_true(Eq::congr(is_sized, e)))

2408// ---- a chunked body ------------------------------------------------------------------
2409//
2410//   chunked-body = *chunk last-chunk trailer-section CRLF
2411//   chunk        = chunk-size [ chunk-ext ] CRLF chunk-data CRLF
2412//   last-chunk   = 1*"0" [ chunk-ext ] CRLF
2413//
2414// The data of each chunk passes on as it arrives; extensions and trailers are
2415// read past and dropped. `room` bounds each size line (with its extension)
2416// and the trailer section, so neither can grow without end.
2417
2418pub(open) inductive Chunks: Type {
2419  // A chunk's size: the value so far, whether a digit came, room left on the line.
2420  ChunkSize(n: Nat, any: Bool, room: Nat): Chunks
2421  // A chunk extension, up to the CR.
2422  ChunkExt(n: Nat, room: Nat): Chunks
2423  // The size line's CR came.
2424  ChunkSizeLf(n: Nat): Chunks
2425  // Bytes of data still to come; at zero, the CR after the data.
2426  ChunkData(left: Nat): Chunks
2427  // The CR after the data came.
2428  ChunkDataLf: Chunks
2429  // At the start of a trailer line, with room left for the trailers.
2430  TrailerStart(room: Nat): Chunks
2431  TrailerLine(room: Nat): Chunks
2432  TrailerLf(room: Nat): Chunks
2433  // The CR of the empty line that ends the body came.
2434  ChunksEndLf: Chunks
2435  // The body ended.
2436  ChunksDone: Chunks
2437  // The body is malformed or did not fit its room.
2438  ChunksBad: Chunks
2439}

2581  pub fn step(line: Nat, c: Chunks, b: Nat): Option(@state Chunks × @rest List(Nat)) {
2582    match c {
2583      ChunkSize(n, any, room) => ⊢ Option(@state Chunks × @rest List(Nat)) from match room {

2590      }
2591      ChunkSizeLf(n) => took(size_end(Nat::eq(b, 10), n, line))
2592      ChunkData(left) => ⊢ Option(@state Chunks × @rest List(Nat)) from match left {
2593        Zero => took(expect(Nat::eq(b, 13), ChunkDataLf))
