6// The small pieces on their own: a service made from a function, the three

9fn double(n: Nat): Outcome(Nat, Nat) {
10  Outcome::Ok(<Res: Nat, E: Nat>, add(n, n))
11}
12
13fn base(): Service(Unit, Nat, Nat, Nat) {
14  service_fn(double)
15}
16
17fn add(a: Nat, b: Nat): Nat {
18  match a {
19    Zero => b
20    Succ(k) => Succ(add(k, b))
21  }
22}

64// Zero is refused with the error `Zero`; anything else passes.
65fn not_zero(n: Nat): Option(Nat) {
66  match n {
67    Zero => Some(Zero)
68    Succ(_) => None
69  }
70}

96// The laws `spire` proves hold for the layers made here: the identity layer changes
97// nothing (`Layer::identity_left`), and stacking is associative (`Layer::then_assoc`).
98fn request_layer(): Layer(Service(Unit, Nat, Nat, Nat), Service(Unit, Nat, Nat, Nat)) {
99  map_request(<S: Unit, R1: Nat, R2: Nat, Res: Nat, E: Nat>, inc)
100}
101
102theorem identity_changes_nothing(): Layer::then(
103  Layer::identity(),
104  request_layer()
105) = request_layer() {
106  Layer::identity_left(request_layer())
107}
108
109theorem stacking_is_associative(): Layer::then(
110  Layer::then(request_layer(), request_layer()),
111  request_layer()
112) = Layer::then(request_layer(), Layer::then(request_layer(), request_layer())) {
113  Layer::then_assoc(request_layer(), request_layer(), request_layer())
114}
