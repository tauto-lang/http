6// A flaky backend: its state counts the calls, and it answers only every
7// second call. Retrying once gets an answer; the rate limit allows two
8// requests until the next refill.
9
10inductive FlakyError: Type {
11  Busy: FlakyError
12}
13
14fn is_even(n: Nat): Bool {
15  match n {
16    Zero => true
17    Succ(k) => not(is_even(k))
18  }
19}
20
21fn flaky(): Service(Nat, Nat, Nat, FlakyError) {
22  Service::Mk((calls: Nat, r: Nat) => ⊢ @outcome Outcome(Nat, FlakyError) × @state Nat from match is_even(calls) {
23    true => #(@outcome Outcome::Err(<Res: Nat, E: FlakyError>, Busy), @state Succ(calls))
24    false => #(@outcome Outcome::Ok(<Res: Nat, E: FlakyError>, Succ(r)), @state Succ(calls))
25  })
26}
27
28fn always(r: Nat, e: FlakyError): Bool {
29  true
30}
31
32// Retry inside, rate limit outside.
33fn stack(): Service(@tokens Nat × @state Nat, Nat, Nat, Limited(FlakyError)) {
34  Layer::apply(Layer::then(retry(always, Succ(Zero)), rate_limit()), flaky())
35}
36
37theorem stack_answers(): outcomes_of(
38  run_all(
39    stack(),
40    Cons(Zero, Cons(Succ(Zero), Cons(Zero, Nil))),
41    #(@tokens Succ(Succ(Zero)), @state Zero)
42  )
43) = Cons(
44  Outcome::Ok(<Res: Nat, E: Limited(FlakyError)>, Succ(Zero)),
45  Cons(
46    Outcome::Ok(<Res: Nat, E: Limited(FlakyError)>, Succ(Succ(Zero))),
47    Cons(Outcome::Err(<Res: Nat, E: Limited(FlakyError)>, Overloaded), Nil)
48  )
49) {
50  Refl
51}
