4// ---- request, response and error mapping ----------------------------------
5
6pub fn map_request<S, R1, R2, Res, E>(
7  f: (R1) -> R2
8): Layer(Service(S, R2, Res, E), Service(S, R1, Res, E)) {
9  Layer::Mk(
10    (svc: Service(S, R2, Res, E)) => Service::Mk((s: S, r: R1) => service_call(svc, s, f(r)))
11  )
12}
13
14pub fn outcome_map<Res, Res2, E, E2>(
15  f: (Res) -> Res2,
16  g: (E) -> E2,
17  o: Outcome(Res, E)
18): Outcome(Res2, E2) {
19  match o {
20    Ok(r) => Ok(f(r))
21    Err(e) => Err(g(e))
22  }
23}
24
25fn map_outcome_pair<S, Res, Res2, E, E2>(
26  f: (Res) -> Res2,
27  g: (E) -> E2,
28  p: @outcome Outcome(Res, E) × @state S
29): @outcome Outcome(Res2, E2) × @state S {
30  match p {
31    #(@outcome o, @state s) => #(@outcome outcome_map(f, g, o), @state s)
32  }
33}
34
35pub fn map_response<S, Req, Res, Res2, E>(
36  f: (Res) -> Res2
37): Layer(Service(S, Req, Res, E), Service(S, Req, Res2, E)) {
38  Layer::Mk(
39    (svc: Service(S, Req, Res, E)) => Service::Mk(
40      (s: S, r: Req) => map_outcome_pair(f, (e: E) => e, service_call(svc, s, r))
41    )
42  )
43}
44
45pub fn map_err<S, Req, Res, E, E2>(
46  g: (E) -> E2
47): Layer(Service(S, Req, Res, E), Service(S, Req, Res, E2)) {
48  Layer::Mk(
49    (svc: Service(S, Req, Res, E)) => Service::Mk(
50      (s: S, r: Req) => map_outcome_pair((x: Res) => x, g, service_call(svc, s, r))
51    )
52  )
53}

55// Mapping requests twice is mapping by the composite once.
56pub theorem map_request_then<S: Type, R1: Type, R2: Type, R3: Type, Res: Type, E: Type>(
57  f: (R1) -> R2,
58  g: (R2) -> R3,
59  svc: Service(S, R3, Res, E),
60  s: S,
61  r: R1
62): service_call(
63  Layer::apply(Layer::then(map_request(g), map_request(f)), svc),
64  s,
65  r
66) = service_call(Layer::apply(map_request((x: R1) => g(f(x))), svc), s, r) {
67  Refl
68}

72// `check` says why a request is refused, or nothing. A refused request never
73// reaches the service and leaves the state as it was.
74fn filter_step<S, Req, Res, E>(
75  svc: Service(S, Req, Res, E),
76  s: S,
77  r: Req,
78  o: Option(E)
79): @outcome Outcome(Res, E) × @state S {
80  match o {
81    None => service_call(svc, s, r)
82    Some(e) => #(@outcome Outcome::Err(<Res: Res, E: E>, e), @state s)
83  }
84}

86pub fn filter<S, Req: Shared, Res, E>(
87  check: (Req) -> Option(E)
88): Layer(Service(S, Req, Res, E), Service(S, Req, Res, E)) {
89  Layer::Mk(
90    (svc: Service(S, Req, Res, E)) => Service::Mk(
91      (s: S, r: Req) => filter_step(svc, s, r, check(r))
92    )
93  )
94}

96pub theorem filter_refused<S: Type, Req: Type, Res: Type, E: Type>(
97  check: (Req) -> Option(E),
98  svc: Service(S, Req, Res, E),
99  s: S,
100  r: Req,
101  e: E,
102  h: check(r) = Some(e)
103): service_call(
104  Layer::apply(filter(check), svc),
105  s,
106  r
107) = #(@outcome Outcome::Err(<Res: Res, E: E>, e), @state s) {
108  Eq::subst(
109    (o: Option(E)) => filter_step(
110      <S: S, Req: Req, Res: Res, E: E>,
111      svc,
112      s,
113      r,
114      o
115    ) = #(@outcome Outcome::Err(<Res: Res, E: E>, e), @state s),
116    Eq::sym(h),
117    Refl
118  )
119}

121// A request the check lets through is the service's own call.
122pub theorem filter_passed<S: Type, Req: Type, Res: Type, E: Type>(
123  check: (Req) -> Option(E),
124  svc: Service(S, Req, Res, E),
125  s: S,
126  r: Req,
127  h: check(r) = None
128): service_call(Layer::apply(filter(check), svc), s, r) = service_call(svc, s, r) {
129  Eq::subst(
130    (o: Option(E)) => filter_step(
131      <S: S, Req: Req, Res: Res, E: E>,
132      svc,
133      s,
134      r,
135      o
136    ) = service_call(svc, s, r),
137    Eq::sym(h),
138    Refl
139  )
