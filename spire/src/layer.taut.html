1// A layer wraps one service in another. For the middleware here `A` and `B`
2// are service types, `Service(S, Req, Res, E)` to `Service(S2, Req2, Res2, E2)`,
3// but composing layers does not look inside them.
4pub(open) inductive Layer(A: Type, B: Type): Type {
5  Mk(wrap: (A) -> B): Layer(A, B)
6}
7
8pub module Layer {
9  pub fn apply<A, B>(l: Layer(A, B), a: A): B {
10    match l {
11      Mk(w) => w(a)
12    }
13  }
14
15  // `inner` first, then `outer` around it.
16  pub fn then<A, B, C>(inner: Layer(A, B), outer: Layer(B, C)): Layer(A, C) {
17    Layer::Mk((a: A) => Layer::apply(outer, Layer::apply(inner, a)))
18  }
19
20  pub fn identity<A>(): Layer(A, A) {
21    Layer::Mk((a: A) => a)
22  }
23
24  // A layer is the same as rebuilding it from its wrap.
25  theorem eta<A: Type, B: Type>(l: Layer(A, B)): Layer::Mk((a: A) => Layer::apply(l, a)) = l {
26    match l {
27      Mk(w) => Refl
28    }
29  }
30
31  pub theorem then_assoc<A: Type, B: Type, C: Type, D: Type>(
32    f: Layer(A, B),
33    g: Layer(B, C),
34    h: Layer(C, D)
35  ): Layer::then(Layer::then(f, g), h) = Layer::then(f, Layer::then(g, h)) {
36    Refl
37  }
38
39  pub theorem identity_left<A: Type, B: Type>(
40    f: Layer(A, B)
41  ): Layer::then(Layer::identity(), f) = f {
42    Layer::eta(f)
43  }
44
45  pub theorem identity_right<A: Type, B: Type>(
46    f: Layer(A, B)
47  ): Layer::then(f, Layer::identity()) = f {
48    Layer::eta(f)
49  }
50
51  // Applying a composite is applying the parts, inner first.
52  theorem then_apply<A: Type, B: Type, C: Type>(
53    f: Layer(A, B),
54    g: Layer(B, C),
55    a: A
56  ): Layer::apply(Layer::then(f, g), a) = Layer::apply(g, Layer::apply(f, a)) {
57    Refl
58  }
59}
