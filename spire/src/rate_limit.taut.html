5// Rate limit: the layer keeps a number of tokens next to the service's state.
6// A request takes one token and goes to the service; with no token left it is
7// refused as `overloaded` and the service is not called. `refill` puts the
8// tokens back, so calling it once per period allows `n` requests per period.
9
10pub(open) inductive Limited(E: Type): Type {
11  Overloaded: Limited(E)
12  Inner(e: E): Limited(E)
13}
14
15fn limited_inner<S, Res, E>(
16  t: Nat,
17  p: @outcome Outcome(Res, E) × @state S
18): @outcome Outcome(Res, Limited(E)) × @state (@tokens Nat × @state S) {
19  match p {
20    #(@outcome o, @state s) => #(
21      @outcome outcome_map((x: Res) => x, (e: E) => Inner(e), o),
22      @state #(@tokens t, @state s)
23    )
24  }
25}
26
27fn rate_limit_step<S, Req, Res, E>(
28  svc: Service(S, Req, Res, E),
29  st: @tokens Nat × @state S,
30  r: Req
31): @outcome Outcome(Res, Limited(E)) × @state (@tokens Nat × @state S) {
32  match st {
33    #(@tokens n, @state s) => ⊢ @outcome Outcome(
34      Res,
35      Limited(E)
36    ) × @state (@tokens Nat × @state S) from match n {
37      Zero => #(
38        @outcome Outcome::Err(<Res: Res, E: Limited(E)>, Overloaded),
39        @state #(@tokens Zero, @state s)
40      )
41      Succ(t) => limited_inner(t, service_call(svc, s, r))
42    }
43  }
44}

46pub fn rate_limit<S, Req, Res, E>(): Layer(
47  Service(S, Req, Res, E),
48  Service(@tokens Nat × @state S, Req, Res, Limited(E))
49) {
50  Layer::Mk(
51    (svc: Service(S, Req, Res, E)) => Service::Mk(
52      (st: @tokens Nat × @state S, r: Req) => rate_limit_step(svc, st, r)
53    )
54  )
55}

57// The start of a period: `n` tokens again.
58pub fn refill<S>(n: Nat, st: @tokens Nat × @state S): @tokens Nat × @state S {
59  match st {
60    #(@tokens t, @state s) => #(@tokens n, @state s)
61  }
62}

64fn tokens<S>(st: @tokens Nat × @state S): Nat {
65  match st {
66    #(t, s) => t
67  }
68}

70// The state a limited service leaves.
71fn limited_state<O, S>(p: @outcome O × @state (@tokens Nat × @state S)): @tokens Nat × @state S {
72  match p {
73    #(o, st) => st
74  }

77// The state at the end of `run_all`.
78pub fn run_state<O, T>(p: @outcomes O × @state T): T {
79  match p {
80    #(os, st) => st

84// The answers `run_all` gave.

91// One token fewer after an admitted request, whatever the service answers.
92theorem limited_inner_tokens<S: Type, Res: Type, E: Type>(
93  t: Nat,
94  p: @outcome Outcome(Res, E) × @state S
95): tokens(limited_state(limited_inner(t, p))) = t {
96  match p {
97    #(@outcome o, @state s) => Refl
98  }
99}

112// With no token, a request is refused, and neither the tokens nor the service's state change.
113theorem rate_limit_empty<S: Type, Req: Type, Res: Type, E: Type>(

116  r: Req

130fn run_prepend<O: Shared, S: Shared>(
131  o: O,
132  rest: @outcomes List(O) × @state S
133): @outcomes List(O) × @state S {
134  #(@outcomes Cons(o, outcomes_of(rest)), @state run_state(rest))
135}

137fn run_with<O: Shared, S: Shared>(
138  p: @outcome O × @state S,
139  k: (S) -> @outcomes List(O) × @state S
140): @outcomes List(O) × @state S {
141  match p {
142    #(@outcome o, @state s) => run_prepend(o, k(s))
143  }
144}

146// The answers to `reqs`, called one after another, and the state at the end.
147pub fn run_all<S: Shared, Req, Res: Shared, E: Shared>(
148  svc: Service(S, Req, Res, E),
149  reqs: List(Req),
150  s: S
151): @outcomes List(Outcome(Res, E)) × @state S {
152  match reqs {
153    Nil => #(@outcomes Nil, @state s)
154    Cons(r, rest) => match service_call(svc, s, r) {
155      #(@outcome o, @state s1) => run_prepend(o, run_all(svc, rest, s1))
156    }
157  }
158}

160// ---- at most `n` admitted requests between refills -------------------------
161
162fn spire_add(a: Nat, b: Nat): Nat {
163  match a {
164    Zero => b
165    Succ(k) => Succ(spire_add(k, b))
166  }
167}

169// How many answers the service gave itself, a response or its own error.
170fn admitted<Res, E>(os: List(Outcome(Res, Limited(E)))): Nat {
171  match os {
172    Nil => Zero
173    Cons(o, rest) => ⊢ Nat from match o {
174      Outcome::Ok(x) => Succ(admitted(rest))
175      Outcome::Err(e) => ⊢ Nat from match e {
176        Overloaded => admitted(rest)
177        Inner(e2) => Succ(admitted(rest))
178      }
179    }
180  }
181}

183theorem rate_limit_admitted_step<S: Type, Res: Type, E: Type>(
184  t: Nat,
185  q: @outcome Outcome(Res, E) × @state S,
186  k: (@tokens Nat × @state S) -> @outcomes List(
187    Outcome(Res, Limited(E))
188  ) × @state (@tokens Nat × @state S),
189  ih: ∀ s2: S. (
190    spire_add(
191      admitted(outcomes_of(k(#(@tokens t, @state s2)))),
192      tokens(run_state(k(#(@tokens t, @state s2))))
193    ) = t
194  )
195): spire_add(
196  admitted(outcomes_of(run_with(limited_inner(t, q), k))),
197  tokens(run_state(run_with(limited_inner(t, q), k)))
198) = Succ(t) {
199  ⊢ spire_add(
200    admitted(outcomes_of(run_with(limited_inner(t, q), k))),
201    tokens(run_state(run_with(limited_inner(t, q), k)))
202  ) = Succ(t) from match q {
203    #(@outcome o, @state s) => ⊢ spire_add(
204      admitted(outcomes_of(run_with(limited_inner(t, #(@outcome o, @state s)), k))),
205      tokens(run_state(run_with(limited_inner(t, #(@outcome o, @state s)), k)))
206    ) = Succ(t) from match o {
207      Outcome::Ok(x) => Eq::congr(Succ, ih(s))
208      Outcome::Err(e) => Eq::congr(Succ, ih(s))
209    }
210  }
211}

213// Starting with `n` tokens, the admitted requests and the tokens left add up
214// to `n`: however many requests come, at most `n` reach the service.
215pub theorem rate_limit_budget<S: Type, Req: Type, Res: Type, E: Type>(
216  svc: Service(S, Req, Res, E),
217  reqs: List(Req),
218  n: Nat,
219  s: S
220): spire_add(
221  admitted(outcomes_of(run_all(Layer::apply(rate_limit(), svc), reqs, #(@tokens n, @state s)))),
222  tokens(run_state(run_all(Layer::apply(rate_limit(), svc), reqs, #(@tokens n, @state s))))
223) = n {
224  ⊢ spire_add(
225    admitted(outcomes_of(run_all(Layer::apply(rate_limit(), svc), reqs, #(@tokens n, @state s)))),
226    tokens(run_state(run_all(Layer::apply(rate_limit(), svc), reqs, #(@tokens n, @state s))))
227  ) = n from match reqs {
228    Nil => Refl
229    Cons(r, rest) => ⊢ spire_add(
230      admitted(
231        outcomes_of(run_all(Layer::apply(rate_limit(), svc), Cons(r, rest), #(@tokens n, @state s)))
232      ),
233      tokens(
234        run_state(run_all(Layer::apply(rate_limit(), svc), Cons(r, rest), #(@tokens n, @state s)))
235      )
236    ) = n from match n {
237      Zero => rate_limit_budget(svc, rest, Zero, s)
238      Succ(t) => rate_limit_admitted_step(
239        t,
240        service_call(svc, s, r),
241        (st: @tokens Nat × @state S) => run_all(Layer::apply(rate_limit(), svc), rest, st),
242        (s2: S) => rate_limit_budget(svc, rest, t, s2)
243      )
244    }
245  }
