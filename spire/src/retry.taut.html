4// Retry: on an error that `should` accepts, call the service again with the
5// state the failed call left, at most `n` more times. The first answer that
6// is a response, or the last error, is the result.
7
8// `stop` on `false`, `go` on `true`.
9fn yn_pick<A>(y: Bool, stop: A, go: A): A {
10  match y { false => stop, true => go }
11}
12
13theorem yn_pick_no<A: Type>(y: Bool, stop: A, go: A, h: y = false): yn_pick(y, stop, go) = stop {
14  Eq::subst((z: Bool) => yn_pick(<A: A>, z, stop, go) = stop, Eq::sym(h), Refl)
15}
16
17// What to do with the answer of one attempt, `left` attempts still allowed.
18fn retry_after<S: Shared, Req: Shared, Res, E: Shared>(
19  svc: Service(S, Req, Res, E),
20  should: (Req, E) -> Bool,
21  left: Nat,
22  r: Req,
23  p: @outcome Outcome(Res, E) × @state S
24): @outcome Outcome(Res, E) × @state S {
25  match left {
26    Zero => p
27    Succ(m) => ⊢ @outcome Outcome(Res, E) × @state S from match p {
28      #(@outcome o, @state s) => ⊢ @outcome Outcome(Res, E) × @state S from match o {
29        Outcome::Ok(x) => #(@outcome Outcome::Ok(<Res: Res, E: E>, x), @state s)
30        Outcome::Err(e) => yn_pick(
31          should(r, e),
32          #(@outcome Outcome::Err(<Res: Res, E: E>, e), @state s),
33          retry_after(svc, should, m, r, service_call(svc, s, r))
34        )
35      }
36    }
37  }
38}

40pub fn retry<S: Shared, Req: Shared, Res, E: Shared>(
41  should: (Req, E) -> Bool,
42  n: Nat
43): Layer(Service(S, Req, Res, E), Service(S, Req, Res, E)) {
44  Layer::Mk(
45    (svc: Service(S, Req, Res, E)) => Service::Mk(
46      (s: S, r: Req) => retry_after(svc, should, n, r, service_call(svc, s, r))
47    )
48  )
49}

51// No retries is the service itself.
52pub theorem retry_zero<S: Type, Req: Type, Res: Type, E: Type>(
53  should: (Req, E) -> Bool,
54  svc: Service(S, Req, Res, E),
55  s: S,
56  r: Req
57): service_call(Layer::apply(retry(should, Zero), svc), s, r) = service_call(svc, s, r) {
58  Refl
59}

61// A response is never retried.
62pub theorem retry_after_ok<S: Type, Req: Type, Res: Type, E: Type>(
63  svc: Service(S, Req, Res, E),
64  should: (Req, E) -> Bool,
65  n: Nat,
66  r: Req,
67  x: Res,
68  s: S
69): retry_after(
70  svc,
71  should,
72  n,
73  r,
74  #(@outcome Outcome::Ok(<Res: Res, E: E>, x), @state s)
75) = #(@outcome Outcome::Ok(<Res: Res, E: E>, x), @state s) {
76  ⊢ retry_after(
77    svc,
78    should,
79    n,
80    r,
81    #(@outcome Outcome::Ok(<Res: Res, E: E>, x), @state s)
82  ) = #(@outcome Outcome::Ok(<Res: Res, E: E>, x), @state s) from match n {
83    Zero => Refl
84    Succ(m) => Refl
85  }
86}

88// An error `should` refuses is never retried.
89pub theorem retry_after_refused<S: Type, Req: Type, Res: Type, E: Type>(
90  svc: Service(S, Req, Res, E),
91  should: (Req, E) -> Bool,
92  n: Nat,
93  r: Req,
94  e: E,
95  s: S,
96  h: should(r, e) = false
97): retry_after(
98  svc,
99  should,
100  n,
101  r,
102  #(@outcome Outcome::Err(<Res: Res, E: E>, e), @state s)
103) = #(@outcome Outcome::Err(<Res: Res, E: E>, e), @state s) {
104  ⊢ retry_after(
105    svc,
106    should,
107    n,
108    r,
109    #(@outcome Outcome::Err(<Res: Res, E: E>, e), @state s)
110  ) = #(@outcome Outcome::Err(<Res: Res, E: E>, e), @state s) from match n {
111    Zero => Refl
112    Succ(m) => yn_pick_no(
113      should(r, e),
114      #(@outcome Outcome::Err(<Res: Res, E: E>, e), @state s),
115      retry_after(svc, should, m, r, service_call(svc, s, r)),
116      h
117    )
118  }
