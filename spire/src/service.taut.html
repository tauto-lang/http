1// What a call gives back: a response, or an error.
2pub(open) inductive Outcome(Res: Type, E: Type): Type {
3  Ok(r: Res): Outcome(Res, E)
4  Err(e: E): Outcome(Res, E)
5}
6
7// A service answers a request with a response or an error and moves its state
8// along: `(S, Req) -> @outcome Outcome(Res, E) × @state S`. A service with nothing to remember
9// uses `S = ()`. Time and capacity are part of the state, so a middleware that
10// counts or refills is an ordinary function and its behaviour can be proved.
11pub(open) inductive Service(S: Type, Req: Type, Res: Type, E: Type): Type {
12  Mk(call: (S, Req) -> @outcome Outcome(Res, E) × @state S): Service(S, Req, Res, E)
13}
14
15pub fn service_call<S, Req, Res, E>(
16  svc: Service(S, Req, Res, E),
17  s: S,
18  r: Req
19): @outcome Outcome(Res, E) × @state S {
20  match svc {
21    Mk(f) => f(s, r)
22  }
23}
24
25// A service that does not use its state, from a function. The state passes through.
26pub fn service_fn<S, Req, Res, E>(f: (Req) -> Outcome(Res, E)): Service(S, Req, Res, E) {
27  Service::Mk((s: S, r: Req) => #(@outcome f(r), @state s))
28}
